    local function attachPortalEffects(window, options)
        options = options or {}
        local root, scale = window.root, window.scale
        local initiallyVisible = not options.InitiallyHidden
        local destroyed, animating, targetVisible, currentVisible = false, false, initiallyVisible, initiallyVisible
        local pendingDestroy = false
        local position, baseScale, baseRotation = root.Position, 1, 0
        local activeTweens = {}
        local transparencySnapshot = {}
        local interactableSnapshot = {}
        local connections = {}
        local originalDestroy = window.Destroy
        local readyAt = os.clock() + 0.34

        local function enforceClipping()
            root.Rotation = 0
            root.ClipsDescendants = true

            for _, descendant in ipairs(root:GetDescendants()) do
                if descendant:IsA("ScrollingFrame") then
                    descendant.ClipsDescendants = true
                end
            end
        end

        enforceClipping()

        screenGui.Name = "VeteranoPortalFX"
        screenGui.ResetOnSpawn = false
        screenGui.IgnoreGuiInset = true
        screenGui.DisplayOrder = window.gui.DisplayOrder + 1
        screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        screenGui.Parent = window.gui.Parent
        window.maid:Add(screenGui)

        window.maid:Add(function()
            destroyed = true

            for _, value in ipairs(activeTweens) do
                value:Cancel()
            end

            table.clear(activeTweens)

            for key in pairs(connections) do
                key:Disconnect()
            end

            table.clear(connections)
        end)

        local function cancelTweens()
            for _, value in ipairs(activeTweens) do
                pcall(function()
                    value:Cancel()
                end)
            end

            table.clear(activeTweens)
        end

        local function playTween(instance, duration, goal, easingStyle, easingDirection)
            local tweenObject = TweenService:Create(instance, TweenInfo.new(duration, easingStyle or Enum.EasingStyle.Quint, easingDirection or Enum.EasingDirection.Out), goal)
            table.insert(activeTweens, tweenObject)
            tweenObject:Play()
            return tweenObject
        end

        local function captureTransparency()
            table.clear(transparencySnapshot)
            local objects = root:GetDescendants()
            table.insert(objects, root)

            for _, value in ipairs(objects) do
                local props = {}

                if value:IsA("GuiObject") then
                    props.BackgroundTransparency = value.BackgroundTransparency
                end

                if value:IsA("TextLabel") or value:IsA("TextButton") or value:IsA("TextBox") then
                    props.TextTransparency = value.TextTransparency
                    props.TextStrokeTransparency = value.TextStrokeTransparency
                end

                if value:IsA("ImageLabel") or value:IsA("ImageButton") then
                    props.ImageTransparency = value.ImageTransparency
                end

                if value:IsA("UIStroke") then
                    props.Transparency = value.Transparency
                end

                if next(props) then
                    transparencySnapshot[value] = props
                end
            end
        end

        local function fadeAll(alpha, duration, style, direction)
            for key, value in pairs(transparencySnapshot) do
                if key.Parent then
                    local goal = {}

                    for key2, value2 in pairs(value) do
                        goal[key2] = value2 + (1 - value2) * alpha
                    end

                    if duration == 0 then
                        for key2, value2 in pairs(goal) do
                            key[key2] = value2
                        end
                    else
                        playTween(key, duration, goal, style or Enum.EasingStyle.Sine, direction)
                    end
                end
            end
        end

        local function disableInteraction()
            table.clear(interactableSnapshot)

            for _, descendant in ipairs(root:GetDescendants()) do
                if descendant:IsA("GuiObject") then
                    local success, result = pcall(function()
                        return descendant.Interactable
                    end)

                    if success then
                        interactableSnapshot[descendant] = result

                        pcall(function()
                            descendant.Interactable = false
                        end)
                    end
                end
            end
        end

        local function restoreInteraction()
            for key, value in pairs(interactableSnapshot) do
                if key.Parent then
                    pcall(function()
                        key.Interactable = value
                    end)
                end
            end

            table.clear(interactableSnapshot)
        end

        local function spawnBurst(origin, implode, duration)

            frame.BackgroundTransparency = 1
            frame.Size = UDim2.fromOffset(1, 1)
            frame.Position = UDim2.fromOffset(origin.X, origin.Y)
            frame.Parent = screenGui
            Debris:AddItem(frame, duration + 0.65)

            for i = 1, 3 do

                frame2.AnchorPoint = Vector2.new(0.5, 0.5)
                frame2.BackgroundTransparency = 1
                local startSize = implode and (140 + i * 28) or (10 + i * 4)
                frame2.Size = UDim2.fromOffset(startSize, startSize)

                frame2.Parent = frame
                Instance.new("UICorner", frame2).CornerRadius = UDim.new(1, 0)

                uiStroke.Thickness = i == 1 and 2.5 or 1.2

                    NumberSequenceKeypoint.new(0.4, 0),
                    NumberSequenceKeypoint.new(0.8, 0.2),
                    NumberSequenceKeypoint.new(1, 1),
                })

                uiGradient.Parent = uiStroke

                playTween(frame2, duration, {
                    Size = UDim2.fromOffset(endSize, endSize),
                    Rotation = frame2.Rotation + (implode and 170 or -120),
                }, Enum.EasingStyle.Quart, implode and Enum.EasingDirection.In or Enum.EasingDirection.Out)

                playTween(uiStroke, duration, {Transparency = 1})
            end

            local frame2 = Instance.new("Frame")
            frame2.AnchorPoint = Vector2.new(0.5, 0.5)
            frame2.Size = UDim2.fromOffset(implode and 16 or 32, implode and 16 or 32)
            frame2.BackgroundColor3 = Color3.fromRGB(235, 216, 255)
            frame2.BackgroundTransparency = 0.12
            frame2.BorderSizePixel = 0
            frame2.Parent = frame
            Instance.new("UICorner", frame2).CornerRadius = UDim.new(1, 0)
            playTween(frame2, duration, {Size = UDim2.fromOffset(2, 2), BackgroundTransparency = 1})

            for i = 1, 16 do
                local angle = i / 16 * math.pi * 2
                local radius = 65 + (i % 4) * 18
                local offset = Vector2.new(math.cos(angle) * radius, math.sin(angle) * radius)
                local frame3 = Instance.new("Frame")
                frame3.AnchorPoint = Vector2.new(0.5, 0.5)
                frame3.Size = UDim2.fromOffset(i % 3 == 0 and 4 or 2, i % 3 == 0 and 4 or 2)
                frame3.Position = implode and UDim2.fromOffset(offset.X, offset.Y) or UDim2.fromOffset(0, 0)
                frame3.BackgroundColor3 = i % 3 == 0 and Color3.fromRGB(214, 182, 255) or ACCENT_COLOR
                frame3.BackgroundTransparency = 0.1
                frame3.BorderSizePixel = 0
                frame3.Parent = frame

                    Position = UDim2.fromOffset(target.X, target.Y),
                    BackgroundTransparency = 1,
                    Rotation = 130,

        local function lerp(a, b, t)
            return a + (b - a) * t
        end

        local function quadBezier(p0, p1, p2, t)
            return lerp(lerp(p0, p1, t), lerp(p1, p2, t), t)
        end

        local function spawnSingularity(origin, duration)
            local closing = true
            local palette = {bg = BACKGROUND_COLOR, accent = ACCENT_COLOR, bright = Color3.fromRGB(218, 194, 255)}

            local function makeFrame(parent, name, position2, size, color, transparency)
                local frame = Instance.new("Frame")
                frame.Name = name
                frame.Position, frame.Size = position2, size
                frame.BackgroundColor3 = color
                frame.BackgroundTransparency = transparency or 0
                frame.BorderSizePixel = 0
                frame.Parent = parent
                return frame
            end

            local function addCorner(target, radius)
                local uiCorner = Instance.new("UICorner")
                uiCorner.CornerRadius = UDim.new(0, radius)
                uiCorner.Parent = target
            end

            local function addStroke(target, color, transparency, thickness)
                local uiStroke = Instance.new("UIStroke")
                uiStroke.Color, uiStroke.Transparency, uiStroke.Thickness = color, transparency, thickness
                uiStroke.Parent = target
                return uiStroke
            end

            local function addGradient(target, color0, color1, rotation)
                local uiGradient = Instance.new("UIGradient")
                uiGradient.Color = ColorSequence.new(color0, color1)
                uiGradient.Rotation = rotation
                uiGradient.Parent = target
                return uiGradient
            end

            if reducedMotion then
                return
            end

            local container = makeFrame(screenGui, "Singularity", UDim2.fromOffset(origin.X, origin.Y), UDim2.fromOffset(1, 1), palette.bg, 1)
            local lenses = {}

            for i = 1, 4 do
                local lens = makeFrame(container, "Lens", UDim2.fromOffset(0, 0), UDim2.fromOffset(1, 1), palette.bg, 1)
                lens.AnchorPoint = Vector2.new(0.5, 0.5)
                addCorner(lens, 999)
                local lensStroke = addStroke(lens, i == 1 and palette.bright or palette.accent, 0.15 + i * 0.12, i == 1 and 2.5 or 1)
                local lensGradient = addGradient(lensStroke, palette.bright, palette.accent, i * 50)

                lensGradient.Transparency = NumberSequence.new({
                    NumberSequenceKeypoint.new(0, 1),
                    NumberSequenceKeypoint.new(0.4, 0),
                    NumberSequenceKeypoint.new(0.8, 0.3),
                    NumberSequenceKeypoint.new(1, 1),
                })

                lenses[i] = {lens, lensStroke, lensGradient}
            end

            local eventHorizon = makeFrame(container, "EventHorizon", UDim2.fromOffset(0, 0), UDim2.fromOffset(10, 10), palette.bg)
            eventHorizon.AnchorPoint = Vector2.new(0.5, 0.5)
            addCorner(eventHorizon, 999)
            addStroke(eventHorizon, palette.bright, 0.2, 2)
            local sparks = {}

            do
                for i = 1, 28 do
                    local angle = i * 2.399963
                    local radius = 54 + (i % 6) * 20
                    local vector2 = Vector2.new(math.cos(angle) * radius, math.sin(angle) * radius * 0.8)
                    local vector22 = Vector2.new(-vector2.Y * 0.9, vector2.X * 0.9)
                    local spark = makeFrame(container, "Spark", UDim2.fromOffset(0, 0), UDim2.fromOffset(i % 4 == 0 and 4 or 2, i % 4 == 0 and 4 or 2), i % 4 == 0 and palette.bright or palette.accent)
                    spark.AnchorPoint = Vector2.new(0.5, 0.5)
                    addCorner(spark, 4)
                    sparks[i] = {spark, vector2, vector22, (i % 5) * 0.025}
                end
            end

            local elapsed = 0
            local connection

            connection = RunService.RenderStepped:Connect(function(deltaTime)
                if destroyed or window.Destroyed or not container.Parent then
                    connection:Disconnect()
                    connections[connection] = nil
                    return
                end

                elapsed = elapsed + deltaTime
                local progress = math.clamp(elapsed / duration, 0, 1)
                local ease = closing and (1 - progress ^ 3) or (1 - (1 - progress) ^ 3)

                for index, value in ipairs(lenses) do
                    local size = 8 + (110 + index * 28) * ease
                    value[1].Size = UDim2.fromOffset(size, size * (index == 3 and 0.42 or 1))
                    value[1].Rotation = (closing and 1 or -1) * progress * (130 + index * 24) + index * 30
                    value[2].Transparency = math.clamp(0.1 + index * 0.1 + progress ^ 3, 0, 1)
                    value[3].Rotation = progress * 240 + index * 30
                end

                local coreSize = closing and (6 + 24 * math.sin(progress * math.pi)) or (28 * (1 - progress) + 2)
                eventHorizon.Size = UDim2.fromOffset(coreSize, coreSize)
                eventHorizon.BackgroundTransparency = progress ^ 3

                for _, value in ipairs(sparks) do
                    local sparkProgress = math.clamp((progress - value[4]) / (1 - value[4]), 0, 1)
                    local fromPos = closing and value[2] or Vector2.zero
                    local toPos = closing and Vector2.zero or value[2]
                    local pos = quadBezier(fromPos, value[3], toPos, sparkProgress)
                    value[1].Position = UDim2.fromOffset(pos.X, pos.Y)
                    value[1].BackgroundTransparency = math.clamp(sparkProgress * sparkProgress, 0, 1)
                end

                if progress >= 1 then
                    connection:Disconnect()
                    connections[connection] = nil
                    container:Destroy()
                end
            end)

            connections[connection] = true
        end

        local function runTransition(opening)
            cancelTweens()
            enforceClipping()

            if opening then
                root.Position, root.Rotation, scale.Scale = position, baseRotation, baseScale

                if options.CenterOnOpen then
                    options.CenterOnOpen(window)
                end

                position, baseScale, baseRotation = root.Position, scale.Scale, 0

                if not next(transparencySnapshot) then
                    captureTransparency()
                end
            else
                position, baseScale, baseRotation = root.Position, scale.Scale, 0
                captureTransparency()
            end

            local duration = reducedMotion and 0.14 or (opening and 0.58 or 0.65)
            local dockPoint

            if opening then
                root.Position = position
                scale.Scale = baseScale
                root.Rotation = baseRotation
            end

            dockPoint = root.AbsolutePosition + Vector2.new(root.AbsoluteSize.X - 22, 26)
            local center = root.AbsolutePosition + root.AbsoluteSize / 2
            local delta = dockPoint - center
            local collapsedPosition = UDim2.new(position.X.Scale, position.X.Offset + delta.X, position.Y.Scale, position.Y.Offset + delta.Y)
            local collapsedScale = reducedMotion and baseScale * 0.96 or baseScale * 0.025

            if window.overlay then
                window.overlay:CloseAll()
            end

            if window.dock then
                window.dock.Visible = false
            end

            window.gui.Enabled = true
            root.Visible = true
            disableInteraction()

            if opening then
                scale.Scale = collapsedScale
                root.Position = reducedMotion and position or collapsedPosition
                root.Rotation = 0
                fadeAll(1, 0)
                spawnBurst(dockPoint, false, duration)
                playTween(scale, duration, {Scale = baseScale}, reducedMotion and Enum.EasingStyle.Quint or Enum.EasingStyle.Back)
                playTween(root, duration, {Position = position})
                fadeAll(0, duration * 0.6)
            else
                if not reducedMotion then
                    playTween(scale, duration * 0.18, {Scale = baseScale * 1.018})
                    task.wait(duration * 0.18)

                    if destroyed or window.Destroyed then
                        return
                    end

                    duration = duration * 0.82
                end

                spawnSingularity(dockPoint, duration)
                playTween(scale, duration, {Scale = collapsedScale}, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
                playTween(root, duration, {Position = reducedMotion and position or collapsedPosition}, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
                fadeAll(1, duration, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
            end

            task.wait(duration + 0.03)

            if destroyed or window.Destroyed then
                return
            end

            root.Visible = opening
            root.Position = position
            root.Rotation = baseRotation
            scale.Scale = baseScale
            fadeAll(0, 0)
            restoreInteraction()
            enforceClipping()

            if opening then
                window:_clampToViewport()
                window:_showDock(window.activeTab and window.activeTab:Root() or nil, false)
                window:_placeDock()
            end

            table.clear(activeTweens)
        end

        local function requestVisible(visible)
            if destroyed or window.Destroyed then
                return
            end

            targetVisible = visible
            window.Hidden = not visible

            if window.launcher then
                window.launcher:_paint()
            end

            if animating then
                return
            end

            animating = true

            task.spawn(function()
                local success, result = pcall(function()
                    local delay = readyAt - os.clock()

                    if delay > 0 then
                        task.wait(delay)
                    end

                    while not destroyed and not window.Destroyed and currentVisible ~= targetVisible do
                        local target = targetVisible
                        runTransition(target)
                        currentVisible = target
                    end
                end)

                animating = false

                if not success and not destroyed and not window.Destroyed then
                    cancelTweens()

                    for key in pairs(connections) do
                        key:Disconnect()
                    end

                    table.clear(connections)

                    for _, child in ipairs(screenGui:GetChildren()) do
                        child:Destroy()
                    end

                    pcall(function()
                        fadeAll(0, 0)
                    end)

                    restoreInteraction()
                    root.Position, root.Rotation, scale.Scale = position, baseRotation, baseScale
                    enforceClipping()
                    root.Visible, window.Hidden = targetVisible, not targetVisible
                    window.gui.Enabled = true

                    if window.dock then
                        window.dock.Visible = false
                    end

                    currentVisible = targetVisible
                    warn("[VETERANO TEAM] Efeito visual recuperado: " .. tostring(result))
                end

                if pendingDestroy and not destroyed and not window.Destroyed then
                    originalDestroy(window)
                end
            end)
        end

        function window:SetVisible(visible)
            if not pendingDestroy then
                requestVisible(visible ~= false)
            end

            return self
        end

        function window:Close()
            return self:SetVisible(false)
        end

        function window:Toggle()
            return self:SetVisible(self.Hidden)
        end

        local controller = {}

        function controller:Destroy()
            if destroyed or window.Destroyed or pendingDestroy then
                return
            end

            pendingDestroy = true
            requestVisible(false)
        end

        if options.InitiallyHidden then
            window.Hidden, root.Visible, window.gui.Enabled = true, false, true

            if window.dock then
                window.dock.Visible = false
            end

            if window.launcher then
                window.launcher:_paint()
            end
        end

        return controller
    end

    return {
        Slate = Slate,
        Accent = ACCENT_COLOR,
        Discord = DISCORD_INVITE,
        Apply = applyBranding,
        Attach = attachPortalEffects,
        SetReducedMotion = function(enabled)
            reducedMotion = enabled == true
        end,
    }
end)()

        local function captureTransparency()
            table.clear(transparencySnapshot)

                local props = {}

                    props.BackgroundTransparency = value.BackgroundTransparency
                end

                if value:IsA("TextLabel") or value:IsA("TextButton") or value:IsA("TextBox") then
                    props.TextTransparency = value.TextTransparency
                    props.TextStrokeTransparency = value.TextStrokeTransparency
                end

                if value:IsA("ImageLabel") or value:IsA("ImageButton") then
                    props.ImageTransparency = value.ImageTransparency
                end

                if value:IsA("UIStroke") then
                    props.Transparency = value.Transparency
                end

                if next(props) then
                    transparencySnapshot[value] = props
                end
            end
        end

        local function fadeAll(alpha, duration, style, direction)
            for key, value in pairs(transparencySnapshot) do
                if key.Parent then
                    local goal = {}

                    for key2, value2 in pairs(value) do

                    if duration == 0 then
                        for key2, value2 in pairs(goal) do
                            key[key2] = value2
                        end
                    else
                        playTween(key, duration, goal, style or Enum.EasingStyle.Sine, direction)
                    end
                end
            end
        end

    local function hasStaffKeyword(text)
        text = string.lower(tostring(text or ""))

        for _, value in ipairs(STAFF_KEYWORDS) do
            if string.find(text, value, 1, true) then
                return true
            end
        end

        return false
    end

    local function isTruthy(value)
        local valueType = typeof(value)

        if valueType == "boolean" then
            return value
        end

        if valueType == "number" then
            return value > 0
        end

        if valueType == "string" then
            local lowered = string.lower(value)
            return lowered ~= "" and lowered ~= "false" and lowered ~= "0" and lowered ~= "none" and lowered ~= "guest" and lowered ~= "member"
        end

        return value ~= nil
    end

    local function isConfiguredStaff(player)
        local VETERANO_SINTONIA_STAFF_USER_IDS = genv.VETERANO_SINTONIA_STAFF_USER_IDS

        if type(VETERANO_SINTONIA_STAFF_USER_IDS) ~= "table" then
            return false
        end

        return VETERANO_SINTONIA_STAFF_USER_IDS[player.UserId] == true or VETERANO_SINTONIA_STAFF_USER_IDS[tostring(player.UserId)] == true or table.find(VETERANO_SINTONIA_STAFF_USER_IDS, player.UserId) ~= nil
    end

    local function detectStaff(player)
        if isConfiguredStaff(player) then
            return true, "configured"
        end

        if game.CreatorType == Enum.CreatorType.User and player.UserId == game.CreatorId then
            return true, "creator"
        end

        local success, result = pcall(function()
            return player:GetAttributes()
        end)

        if success then
            for key, value in pairs(result) do
                if hasStaffKeyword(key) and isTruthy(value) then
                    return true, "attribute:" .. key
                end
            end
        end

        local success2, result2 = pcall(CollectionService.GetTags, CollectionService, player)

        if success2 then
            for _, value in ipairs(result2) do
                if hasStaffKeyword(value) then
                    return true, "tag:" .. value
                end
            end
        end

        return false, nil
    end

    local function getRelationship(player)
        local entry = Runtime.relationships[player]

        if entry then
            return entry
        end

        entry = {
            ready = false,
            checking = false,
            friend = false,
            admin = false,
            rank = 0,
            role = "",
            source = nil,
        }

        Runtime.relationships[player] = entry
        return entry
    end

    local function refreshRelationship(player)
        if player == localPlayer or player.Parent ~= Players then
            return
        end

        local entry = getRelationship(player)

        if entry.checking then
            return
        end

        entry.checking = true

        task.spawn(function()
            local isFriend = false

            local success, result = pcall(function()
                return localPlayer:IsFriendsWith(player.UserId)
            end)

            if success then
                isFriend = result == true
            end

            local isAdmin, source = detectStaff(player)
            local rank, role = 0, ""

            if game.CreatorType == Enum.CreatorType.Group and game.CreatorId > 0 then
                local success2, result2 = pcall(function()
                    return player:GetRankInGroup(game.CreatorId)
                end)

                if success2 and type(result2) == "number" then
                    rank = result2
                end

                local success3, result3 = pcall(function()
                    return player:GetRoleInGroup(game.CreatorId)
                end)

                if success3 and type(result3) == "string" then
                    role = result3
                end

                if rank >= STAFF_RANK_THRESHOLD then
                    isAdmin, source = true, "group-rank:" .. rank
                elseif hasStaffKeyword(role) then
                    isAdmin, source = true, "group-role:" .. role
                end
            end

            if player.Parent == Players and Runtime.relationships[player] == entry then
                entry.friend = isFriend
                entry.admin = isAdmin
                entry.rank = rank
                entry.role = role
                entry.source = source
                entry.ready = true
                entry.checking = false
            end
        end)
    end

    local function ensureRelationship(player)
        local entry = getRelationship(player)

        if not entry.ready and not entry.checking then
            refreshRelationship(player)
        end

        return entry
    end

    function Relationship.Protected(player)
        local entry = ensureRelationship(player)

        if (State.friendCheck or State.adminCheck) and not entry.ready then
            return true
        end

        if State.adminCheck and entry.admin then
            return true
        end

        if State.friendCheck and entry.friend then
            return true
        end

        return false
    end

    Relationship.Admin = function(player)
        local entry = ensureRelationship(player)
        return entry.ready and entry.admin
    end

    Relationship.Appearance = function(player)
        local entry = ensureRelationship(player)

        if entry.ready and (State.adminCheck and entry.admin) then
            return STAFF_COLOR, " [STAFF]"
        end

        if entry.ready and (State.friendCheck and entry.friend) then
            return FRIEND_COLOR, " [AMIGO]"
        end

        return DEFAULT_COLOR, ""
    end

    local function watchPlayer(player)
        if player == localPlayer then
            return
        end

        refreshRelationship(player)

        local connection = player.AttributeChanged:Connect(function(attribute)
            if hasStaffKeyword(attribute) then
                local entry = Runtime.relationships[player]

                if entry then
                    entry.ready = false
                end

                refreshRelationship(player)
            end
        end)

        Runtime.relationshipConnections[player] = connection
        trackConnection(connection)
    end

    for _, player in ipairs(Players:GetPlayers()) do
        watchPlayer(player)
    end

    trackConnection(Players.PlayerAdded:Connect(watchPlayer))

    trackConnection(Players.PlayerRemoving:Connect(function(player)
        Runtime.relationships[player] = nil
        Runtime.relationshipConnections[player] = nil
    end))

do
    local jumpHumanoid, stateConnection, boostConnection
    local jumpGeneration = 0

    local GROUNDED_STATES = {
        [Enum.HumanoidStateType.Running] = true,
        [Enum.HumanoidStateType.RunningNoPhysics] = true,
        [Enum.HumanoidStateType.Landed] = true,
    }

    local function disconnectBoost()
        if boostConnection then
            boostConnection:Disconnect()
            boostConnection = nil
        end
    end

    local function resetJumpBoost()
        jumpGeneration = jumpGeneration + 1
        disconnectBoost()

        if stateConnection then
            stateConnection:Disconnect()
            stateConnection = nil
        end

        jumpHumanoid = nil
    end

    local function jumpBoostBlocked()
        return Runtime.destroyed or not State.spoofJumpEnabled or State.flyEnabled or State.spoofWalkEnabled or State.ghostEnabled or State.noclipCarEnabled or State.vFlyEnabled or Runtime.travelBusy or Runtime.fixedTeleportBusy or Runtime.pumpkinBelowActive or Runtime.collectionBusy or (Runtime.travel and Runtime.travel.owner) or State.autoColetaEnabled or State.farmPartesEnabled or State.autoFarmLixeiroEnabled or State.autoFarmPescaEnabled or State.autoFarmPumpkinEnabled or Runtime.robberyStarting or (Runtime.robbery and (Runtime.robbery.enabled or Runtime.robbery.paused))
    end

    local function scheduleJumpBoost(humanoid)
        if jumpBoostBlocked() or boostConnection then
            return
        end

        local rootPart, character = getRootPart(), localPlayer.Character

        if not rootPart or rootPart.Anchored or humanoid.Health <= 0 or humanoid.Sit or humanoid.PlatformStand then
            return
        end

        local generation = jumpGeneration

        boostConnection = RunService.Heartbeat:Connect(function()
            disconnectBoost()

            if generation ~= jumpGeneration or jumpHumanoid ~= humanoid or jumpBoostBlocked() then
                return
            end

            if localPlayer.Character ~= character or getRootPart() ~= rootPart or getHumanoid() ~= humanoid then
                return
            end

            if not rootPart.Parent or rootPart.Anchored or humanoid.Health <= 0 or humanoid.Sit or humanoid.PlatformStand then
                return
            end

            local state = humanoid:GetState()

            if state ~= Enum.HumanoidStateType.Jumping and state ~= Enum.HumanoidStateType.Freefall then
                return
            end

            local targetPower = tonumber(State.spoofJumpPower)

            if not targetPower or targetPower ~= targetPower or math.abs(targetPower) == math.huge then
                return
            end

            targetPower = math.clamp(targetPower, 0, 65)
            local assemblyLinearVelocity, assemblyMass = rootPart.AssemblyLinearVelocity, rootPart.AssemblyMass
            local deltaVelocity = targetPower - assemblyLinearVelocity.Y

            if deltaVelocity <= 0 or assemblyMass <= 0 or assemblyMass == math.huge then
                return
            end

            local success, result = pcall(function()
                rootPart:ApplyImpulse(Vector3.new(0, deltaVelocity * assemblyMass, 0))
            end)

            if not success then
                warn("[VETERANO TEAM] Pulo: " .. tostring(result))
            end
        end)
    end

    function Runtime.updateJumpPower(humanoid)
        if Runtime.destroyed or not State.spoofJumpEnabled or not humanoid then
            if jumpHumanoid or stateConnection or boostConnection then
                resetJumpBoost()
            end

            return
        end

        if jumpHumanoid == humanoid and stateConnection then
            return
        end

        resetJumpBoost()
        jumpHumanoid = humanoid

        stateConnection = humanoid.StateChanged:Connect(function(oldState, newState)
            if jumpHumanoid ~= humanoid then
                return
            end

            if newState == Enum.HumanoidStateType.Dead then
                disconnectBoost()
                return
            end

            if newState == Enum.HumanoidStateType.Jumping and GROUNDED_STATES[oldState] then
                scheduleJumpBoost(humanoid)
            end
        end)
    end

    function Runtime.setJumpPower(enabled)
        State.spoofJumpEnabled = enabled == true and not Runtime.destroyed
        Runtime.updateJumpPower(getHumanoid())
    end

    Runtime.restoreJumpPower = resetJumpBoost

    Runtime.uz = function()
        Runtime.updateJumpPower(getHumanoid())
    end

    trackConnection(localPlayer.CharacterRemoving:Connect(resetJumpBoost))
end

            if key.Parent then
                pcall(function()
                    key.CanCollide = value
                end)
            end
        end

        table.clear(originalCollide)
    end

    local function noclipWanted()
        return State.noclipEnabled or State.noclipCarEnabled or State.vFlyEnabled or next(noclipOwners) ~= nil
    end

    local function applyNoclip(force)
        local character = localPlayer.Character

        if trackedCharacter ~= character then
            restoreCollisions()
            trackedCharacter = character
            trackCharacterParts(character)
        end

        if Runtime.destroyed or not character or (force ~= true and not noclipWanted()) then
            if next(originalCollide) then
                restoreCollisions()
            end

            return
        end

        for key in pairs(characterParts) do
            if key.Parent then
                if originalCollide[key] == nil then
                    originalCollide[key] = key.CanCollide
                end

                if key.CanCollide then
                    key.CanCollide = false
                end
            end
        end
    end

    Runtime.noclip = {
        acquire = function(owner)
            noclipOwners[owner] = true
            applyNoclip()
        end,
        release = function(owner)
            noclipOwners[owner] = nil
            applyNoclip()
        end,
        step = function()
            applyNoclip()
        end,
        restore = function()
            table.clear(noclipOwners)
            restoreCollisions()

            if Runtime.destroyed then
                trackCharacterParts(nil)
                trackedCharacter = nil
            end
        end,
    }

    noclipStep = function(force)
        applyNoclip(force)
    end

    noclipUpdate = function()
        applyNoclip()
    end

    noop = function() end
    Runtime.sz, Runtime.Nz, Runtime.ez = noclipStep, noclipUpdate, noop

    trackConnection(localPlayer.CharacterAdded:Connect(function()
        applyNoclip()
    end))
end

do
    local FAKE_NAME = "@VETERANO.DEV"
    local spoofedTargets = {}
    local nickTimer = 0

    local function spoofProperty(target, prop)
        if not target or spoofedTargets[target] then
            return
        end

        local success, result = pcall(function()
            return target[prop]
        end)

        if not success or type(result) ~= "string" then
            return
        end

        local entry = {original = result, prop = prop, writing = false}
        spoofedTargets[target] = entry

        local function apply()
            if entry.writing or not State.camuflarNickEnabled or Runtime.destroyed or not target.Parent then
                return
            end

            local current = target[prop]

            if current == FAKE_NAME then
                return
            end

            entry.original = current
            entry.writing = true
            target[prop] = FAKE_NAME
            entry.writing = false
        end

        entry.connection = target:GetPropertyChangedSignal(prop):Connect(function()
            pcall(apply)
        end)

        apply()
    end

    local function refreshNickSpoof()
        for key, value in pairs(spoofedTargets) do
            if not key.Parent then
                value.connection:Disconnect()
                spoofedTargets[key] = nil
            end
        end

        local playerGui = localPlayer:FindFirstChild("PlayerGui")
        local inventoryUI = playerGui and playerGui:FindFirstChild("InventoryUI")
        local invFrame = inventoryUI and inventoryUI:FindFirstChild("InvFrame")
        local invPessoal = invFrame and invFrame:FindFirstChild("InvPessoal")
        local bTxt = invPessoal and invPessoal:FindFirstChild("BTxt")

        if bTxt and (bTxt:IsA("TextLabel") or bTxt:IsA("TextButton") or bTxt:IsA("TextBox")) then
            spoofProperty(bTxt, "Text")
        end

        local humanoid = getHumanoid()

        if humanoid then
            spoofProperty(humanoid, "DisplayName")
        end
    end

    local function restoreNicks()
        for key, value in pairs(spoofedTargets) do
            value.connection:Disconnect()

            if key.Parent then
                pcall(function()
                    key[value.prop] = value.original
                end)
            end
        end

        table.clear(spoofedTargets)
    end

    function Runtime.setCamuflarNick(enabled)
        State.camuflarNickEnabled = enabled == true

        if State.camuflarNickEnabled and not Runtime.destroyed then
            refreshNickSpoof()
        else
            restoreNicks()
        end
    end

    trackConnection(RunService.Heartbeat:Connect(function(deltaTime)
        if not State.camuflarNickEnabled or Runtime.destroyed then
            return
        end

        nickTimer = nickTimer + deltaTime

        if nickTimer >= 0.35 then
            nickTimer = 0
            pcall(refreshNickSpoof)
        end
    end))
end

local function findCarController(target)
    if type(getconnections) ~= "function" then
        return nil
    end

    if type(debug) ~= "table" or type(debug.getupvalue) ~= "function" then
        return nil
    end

    for _, value in ipairs(getconnections(RunService.Stepped)) do
        local function2 = value.Function

        if type(function2) == "function" then
            local matched, controller

            for i = 1, 8 do
                local success, result = pcall(debug.getupvalue, function2, i)

                if not success then
                    break
                end

                if result == target then
                    matched = true
                end

                if type(result) == "table" and result.RemapMultiplier ~= nil and type(result.BuildHPCache) == "function" then
                    controller = result
                end
            end

            if matched and controller then
                return controller
            end
        end
    end

    return nil
end

do
    local carState = {
        carro = nil,
        originais = {},
        raiz = nil,
        assento = nil,
        relativoAssento = nil,
        altura = nil,
        piso = nil,
        raycast = nil,
        velocidade = 0,
        frente = nil,
        personagemAtivo = false,
        altitude = nil,
        vFlyAtivo = false,
        vFlyReduzidoAte = 0,
    }

    local carParts, carPartAdded, carPartRemoved = {}, nil, nil

    local function trackCarParts(car)
        if carPartAdded then
            carPartAdded:Disconnect()
            carPartAdded = nil
        end

        if carPartRemoved then
            carPartRemoved:Disconnect()
            carPartRemoved = nil
        end

        table.clear(carParts)

        if not car then
            return
        end

        for _, descendant in ipairs(car:GetDescendants()) do
            if descendant:IsA("BasePart") then
                carParts[descendant] = true
            end
        end

        carPartAdded = car.DescendantAdded:Connect(function(descendant)
            if descendant:IsA("BasePart") then
                carParts[descendant] = true
            end
        end)

        carPartRemoved = car.DescendantRemoving:Connect(function(descendant)
            carParts[descendant] = nil
        end)
    end

    local function getCarRoot(car)
        local vehicleSeat = car:FindFirstChild("DriveSeat", true) or car:FindFirstChildWhichIsA("VehicleSeat", true)
        return (vehicleSeat and vehicleSeat.AssemblyRootPart) or car.PrimaryPart or vehicleSeat, vehicleSeat
    end

    local function resetCarState()
        trackCarParts(nil)

        for key, value in pairs(carState.originais) do
            if key.Parent then
                pcall(function()
                    key.CanCollide = value
                end)
            end
        end

        table.clear(carState.originais)

        if carState.personagemAtivo and not State.noclipEnabled then
            pcall(noclipStep, false)
        end

        carState.carro, carState.raiz, carState.assento = nil, nil, nil
        carState.relativoAssento, carState.altura, carState.piso, carState.raycast = nil, nil, nil, nil
        carState.velocidade, carState.frente, carState.personagemAtivo = 0, nil, false
        carState.altitude, carState.vFlyAtivo, carState.vFlyReduzidoAte = nil, false, 0
    end

    local function captureCarState(car)
        local root, seat = getCarRoot(car)

        if not root or not seat then
            return
        end

        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        raycastParams.FilterDescendantsInstances = {car, localPlayer.Character}
        local raycastResult = workspace:Raycast(root.Position + Vector3.new(0, 20, 0), Vector3.new(0, -120, 0), raycastParams)
        carState.raiz = root
        carState.assento = seat
        carState.relativoAssento = ((root.CFrame - root.Position)):ToObjectSpace(seat.CFrame - seat.Position)
        carState.raycast = raycastParams
        carState.piso = raycastResult and raycastResult.Position.Y or (root.Position.Y - 3)
        carState.altura = raycastResult and math.max(1, root.Position.Y - raycastResult.Position.Y) or 3
        local vector = Vector3.new(seat.CFrame.LookVector.X, 0, seat.CFrame.LookVector.Z)
        carState.frente = vector.Magnitude > 0 and vector.Unit or Vector3.new(0, 0, -1)
        carState.altitude = root.Position.Y
        carState.velocidade = vector.Magnitude > 0 and root.AssemblyLinearVelocity:Dot(vector.Unit) or 0
    end

    local function approach(current, target, step)
        if current < target then
            return math.min(current + step, target)
        end

        return math.max(current - step, target)
    end

    local function driveCar(deltaTime)
        local raiz, assento = carState.raiz, carState.assento
        local raycast, altura, piso = carState.raycast, carState.altura, carState.piso
        local relativoAssento = carState.relativoAssento

        if not raiz or not raiz.Parent or not assento or not assento.Parent or not raycast or not altura or not piso or not relativoAssento then
            return
        end

        local dt = math.min(deltaTime or 0.016666, 0.1)
        local vFlyEnabled = State.vFlyEnabled
        local correctingHeight = false

        if vFlyEnabled and not carState.vFlyAtivo then
            carState.altitude = raiz.Position.Y
            carState.vFlyReduzidoAte = os.clock() + 0.6
            carState.velocidade = math.clamp(carState.velocidade, -24, 36)
        elseif not vFlyEnabled and carState.vFlyAtivo then
            local raycastResult = workspace:Raycast(raiz.Position + Vector3.new(0, 20, 0), Vector3.new(0, -520, 0), raycast)

            if raycastResult then
                piso = raycastResult.Position.Y
                carState.piso = piso
            end
        end

        carState.vFlyAtivo = vFlyEnabled
        local heightDelta = 0

        if vFlyEnabled then
            if dt >= 0.05 or raiz.Anchored or raiz.AssemblyLinearVelocity.Magnitude < 12 then
                carState.vFlyReduzidoAte = os.clock() + 0.6
            end
        else
            local vector = Vector3.new(raiz.Position.X, piso + 20, raiz.Position.Z)
            local raycastResult = workspace:Raycast(vector, Vector3.new(0, -60, 0), raycast)

            if raycastResult and raycastResult.Position.Y - piso <= 4 then
                piso = raycastResult.Position.Y
                carState.piso = piso
            end

            heightDelta = (piso + altura) - raiz.Position.Y
            correctingHeight = math.abs(heightDelta) > 1

            if correctingHeight then
                heightDelta = math.clamp(heightDelta, -20 * dt, 20 * dt)
            end
        end

        local forward = carState.frente or Vector3.new(0, 0, -1)
        local verticalVelocity = 0

        if forward.Magnitude > 0.001 then
            forward = forward.Unit
            local throttleFloat = assento.ThrottleFloat
            local steerFloat = assento.SteerFloat

            if UserInputService:IsKeyDown(Enum.KeyCode.W) or UserInputService:IsKeyDown(Enum.KeyCode.Up) then
                throttleFloat = 1
            elseif UserInputService:IsKeyDown(Enum.KeyCode.S) or UserInputService:IsKeyDown(Enum.KeyCode.Down) then
                throttleFloat = -1
            end

            if UserInputService:IsKeyDown(Enum.KeyCode.A) or UserInputService:IsKeyDown(Enum.KeyCode.Left) then
                steerFloat = -1
            elseif UserInputService:IsKeyDown(Enum.KeyCode.D) or UserInputService:IsKeyDown(Enum.KeyCode.Right) then
                steerFloat = 1
            end

            local acSilenced = isAcSilenced()
            local boost = math.clamp(State.boostFactor, 1, 999)
            local maxForward

            if vFlyEnabled then
                maxForward = State.vFlyHorizontalSpeed * boost
            else
                maxForward = correctingHeight and 260 or (acSilenced and 999 or 180) * boost
            end

            local maxReverse = vFlyEnabled and (maxForward * 0.65) or (correctingHeight and 120 or (150 * boost))
            local targetSpeed = throttleFloat > 0.05 and maxForward or (throttleFloat < -0.05 and -maxReverse or 0)
            local accel = (math.abs(throttleFloat) > 0.05 and 130 or 200) * dt
            carState.velocidade = approach(carState.velocidade, targetSpeed, accel)

            if math.abs(steerFloat) > 0.05 and math.abs(carState.velocidade) > 0.5 then
                local direction = carState.velocidade >= 0 and 1 or -1
                local turnAngle = ((-steerFloat * direction) * math.rad(85)) * dt
                forward = CFrame.Angles(0, turnAngle, 0):VectorToWorldSpace(forward)
            end

            carState.frente = forward.Unit

            if vFlyEnabled then
                local verticalInput = 0

                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                    verticalInput = 1
                elseif UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                    verticalInput = -1
                end

                local verticalSpeed = State.vFlyVerticalSpeed * boost

                if verticalInput ~= 0 then
                    verticalVelocity = verticalInput * verticalSpeed
                    carState.altitude = raiz.Position.Y + verticalVelocity * dt
                else
                    local altitudeError = (carState.altitude or raiz.Position.Y) - raiz.Position.Y
                    verticalVelocity = math.clamp(altitudeError * 10, -verticalSpeed, verticalSpeed)
                end
            end

            local facing = CFrame.lookAt(Vector3.zero, forward.Unit, Vector3.yAxis)
            local rootRotation = facing * relativoAssento:Inverse()

            if vFlyEnabled then
                raiz.CFrame = CFrame.new(raiz.Position + Vector3.new(0, heightDelta, 0)) * rootRotation
            elseif math.abs(heightDelta) > 0.01 then
                raiz.CFrame = raiz.CFrame + Vector3.new(0, heightDelta, 0)
            end

            if forward.Magnitude > 0.001 then
                raiz.AssemblyLinearVelocity = forward.Unit * carState.velocidade + Vector3.new(0, verticalVelocity, 0)
            else
                raiz.AssemblyLinearVelocity = Vector3.new(0, verticalVelocity, 0)
            end

            raiz.AssemblyAngularVelocity = Vector3.zero
        end
    end

    local function applyCarNoclip()
        if not State.noclipCarEnabled and not State.vFlyEnabled then
            if carState.carro or carState.personagemAtivo or next(carState.originais) then
                resetCarState()
            end

            return
        end

        local car = getCurrentCar()

        if car ~= carState.carro then
            resetCarState()
            carState.carro = car

            if car then
                trackCarParts(car)
                captureCarState(car)
            end
        end

        if not car then
            return
        end

        noclipStep(true)
        carState.personagemAtivo = true

        for key in pairs(carParts) do
            if key.Parent then
                if carState.originais[key] == nil then
                    carState.originais[key] = key.CanCollide
                end

                if key.CanCollide then
                    key.CanCollide = false
                end
            end
        end
    end

    Runtime.aplicarNoclipCar = applyCarNoclip
    Runtime.restaurarNoclipCar = resetCarState

    trackConnection(RunService.Stepped:Connect(function()
        if not Runtime.destroyed then
            pcall(applyCarNoclip)
        end
    end))

    trackConnection(RunService.Heartbeat:Connect(function(deltaTime)
        if not Runtime.destroyed and (State.noclipCarEnabled or State.vFlyEnabled) then
            pcall(driveCar, deltaTime)
        end
    end))
end

local function applyCarBoost(humanoid)
    local controller = findCarController(humanoid)

    if not controller then
        return false
    end

    controller.RemapMultiplier = math.clamp(State.boostFactor, 1, 999)

    pcall(function()
        controller:BuildHPCache()
    end)

    pcall(function()
        if controller.SpeedLimit ~= nil then
            controller.SpeedLimit = 999
        end

        if controller.MaxSpeed ~= nil then
            controller.MaxSpeed = 999
        end

        if controller.TopSpeed ~= nil then
            controller.TopSpeed = 999
        end

        if controller.SpeedCap ~= nil then
            controller.SpeedCap = 999
        end
    end)

    return true
end

    Runtime.vidaDoCarro = function(car)
        local body = car:FindFirstChild("Body")
        local pintura = body and body:FindFirstChild("Pintura")
        local lifeController = pintura and pintura:FindFirstChild("LifeController")
        return lifeController and lifeController:FindFirstChild("Vida")
    end

    local function getSeatedCar()
        local humanoid = getHumanoid()
        local seat = humanoid and humanoid.SeatPart

        if not seat then
            return nil
        end

        local carrosSpawnados = workspace:FindFirstChild("CarrosSpawnados")

        if not carrosSpawnados then
            return nil
        end

        local node = seat

        while node and node.Parent do
            if node.Parent == carrosSpawnados then
                return node
            end

            node = node.Parent
        end

        return nil
    end

    local function unbindCar()
        if durHud.vidaConn then
            durHud.vidaConn:Disconnect()
            durHud.vidaConn = nil
        end

        durHud.car = nil

        if durHud.painel then
            durHud.painel:SetVisible(false)
        end
    end

    local function bindCar(car)
        local healthValue = Runtime.vidaDoCarro(car)

        if not healthValue then
            return
        end

        if durHud.vidaConn then
            durHud.vidaConn:Disconnect()
        end

        durHud.car = car
        durHud.maxVida = math.max(DEFAULT_MAX_HEALTH, healthValue.Value)

        if durHud.carLabel then
            durHud.carLabel.Text = car.Name
        end

        if durHud.painel then
            durHud.painel:SetVisible(true)
        end

        updateHealth(healthValue.Value)

        durHud.vidaConn = healthValue:GetPropertyChangedSignal("Value"):Connect(function()
            updateHealth(healthValue.Value)
        end)
    end

    function Runtime.ativarDurabilidade()
        buildHud()
        durHud.frame.Visible = true
        Runtime.durabilidadeRunId = Runtime.durabilidadeRunId + 1
        local durabilidadeRunId = Runtime.durabilidadeRunId

        task.spawn(function()
            while State.durabilidadeHudEnabled and not Runtime.destroyed and Runtime.durabilidadeRunId == durabilidadeRunId do
                local car = getSeatedCar()

                if car ~= durHud.car then
                    if car and Runtime.vidaDoCarro(car) then
                        pcall(bindCar, car)
                    else
                        pcall(unbindCar)

                        if durHud.frame then
                            durHud.frame.Visible = false
                        end
                    end
                elseif car and not durHud.vidaConn then
                    pcall(bindCar, car)
                end

                if durHud.frame then
                    durHud.frame.Visible = car ~= nil
                end

                task.wait(0.3)
            end
        end)
    end

    function Runtime.desativarDurabilidade()
        Runtime.durabilidadeRunId = Runtime.durabilidadeRunId + 1
        pcall(unbindCar)

        if durHud.frame then
            durHud.frame.Visible = false
        end
    end

    function Runtime.destruirDurabilidade()
        pcall(unbindCar)

        if durHud.gui then
            pcall(function()
                durHud.gui:Destroy()
            end)
        end

        durHud.gui, durHud.frame, durHud.fill = nil, nil, nil
    end
end

    local espOverlay

    local function isFiniteCoord(n)
        return type(n) == "number" and n == n and math.abs(n) < 10000000
    end

    function Runtime.renderESPLine(line, from, to, color, thickness)
        if not line or not line.Parent then
            return false
        end

        if not from or not to or not isFiniteCoord(from.X) or not isFiniteCoord(from.Y) or not isFiniteCoord(to.X) or not isFiniteCoord(to.Y) then
            line.Visible = false
            return false
        end

        local dx, dy = to.X - from.X, to.Y - from.Y
        local length = math.sqrt(dx * dx + dy * dy)

        if length < 0.1 then
            line.Visible = false
            return false
        end

        line.AnchorPoint = Vector2.new(0.5, 0.5)
        line.Position = UDim2.fromOffset((from.X + to.X) * 0.5, (from.Y + to.Y) * 0.5)
        line.Size = UDim2.fromOffset(length, thickness or 1.5)
        line.Rotation = math.deg(math.atan2(dy, dx))
        line.BackgroundColor3 = color
        line.Visible = true
        return true
    end

local DrawingSupport = {medido = false, drawing = false, texto = false}

function DrawingSupport.set(object, prop, value)
    return pcall(function()
        object[prop] = value
    end)
end

function DrawingSupport.medir()
    if DrawingSupport.medido then
        return DrawingSupport
    end

    if Drawing == nil or type(Drawing.new) ~= "function" then
        return DrawingSupport
    end

    local success, result = pcall(Drawing.new, "Square")

    if not success then
        return DrawingSupport
    end

    DrawingSupport.drawing = true
    DrawingSupport.filled = DrawingSupport.set(result, "Filled", false)
    DrawingSupport.thickness = DrawingSupport.set(result, "Thickness", 1)

    pcall(function()
        result:Remove()
    end)

    local success2, result2 = pcall(Drawing.new, "Text")

    if success2 then
        DrawingSupport.texto = true
        DrawingSupport.center = DrawingSupport.set(result2, "Center", true)
        DrawingSupport.outline = DrawingSupport.set(result2, "Outline", true)
        DrawingSupport.outlineColor = DrawingSupport.set(result2, "OutlineColor", Color3.new())
        DrawingSupport.font = DrawingSupport.set(result2, "Font", 2)

        pcall(function()
            result2.Text = "MMM"
        end)

        local success3, result3 = pcall(function()
            return result2.TextBounds
        end)

        DrawingSupport.textBounds = success3 and (typeof(result3) == "Vector2" and result3.X > 0)

        pcall(function()
            result2:Remove()
        end)
    end

    local success3, result3 = pcall(Drawing.new, "Triangle")
    DrawingSupport.triangle = success3 == true

    if success3 then
        pcall(function()
            result3:Remove()
        end)
    end

    DrawingSupport.medido = true
    return DrawingSupport
end

function DrawingSupport.ok()
    return (DrawingSupport.medir()).drawing
end

local function newDrawing(kind)
    if not DrawingSupport.ok() then
        return nil
    end

    if kind == "Triangle" and not DrawingSupport.triangle then
        return nil
    end

    local success, result = pcall(function()
        return Drawing.new(kind)
    end)

    if not success then
        return nil
    end

    DrawingSupport.set(result, "Visible", false)
    DrawingSupport.set(result, "Color", Color3.fromRGB(170, 70, 255))
    DrawingSupport.set(result, "Transparency", 1)
    DrawingSupport.set(result, "Thickness", 1)
    return result
end

local function newText(color)
    if not DrawingSupport.ok() or not DrawingSupport.texto then
        return nil
    end

    local success, result = pcall(function()
        return Drawing.new("Text")
    end)

    if not success then
        return nil
    end

    DrawingSupport.set(result, "Visible", false)
    DrawingSupport.set(result, "Color", color or Color3.fromRGB(170, 70, 255))
    DrawingSupport.set(result, "Transparency", 1)
    DrawingSupport.set(result, "Size", 13)

    if DrawingSupport.center then
        DrawingSupport.set(result, "Center", true)
    end

    if DrawingSupport.outline then
        DrawingSupport.set(result, "Outline", true)
    end

    if DrawingSupport.outlineColor then
        DrawingSupport.set(result, "OutlineColor", Color3.new(0, 0, 0))
    end

    if DrawingSupport.font then
        DrawingSupport.set(result, "Font", 2)
    end

    return result
end

local function textWidth(text)
    if not text then
        return 0
    end

    if DrawingSupport.textBounds then
        local success, result = pcall(function()
            return text.TextBounds
        end)

        if success and result then
            return result.X
        end
    end

    local success, result = pcall(function()
        return text.Text
    end)

    if not success or type(result) ~= "string" then
        return 0
    end

    local success2, result2 = pcall(function()
        return text.Size
    end)

    return (#result * (success2 and result2 or 13)) * 0.52
end

    if last <= 0 or now - last > interval then

function Runtime.prepareESPBundle(bundle, now)
    if now < (bundle.nextDrawingRetry or 0) then
        return
    end

    local failed = false

    if State.box and not bundle.box then
        bundle.box = newDrawing("Square")

        if bundle.box then
            bundle.box.Color = bundle.relationshipColor or Color3.fromRGB(170, 70, 255)

            if DrawingSupport.filled then
                bundle.box.Filled = false
            end
        end

        failed = failed or bundle.box == nil
    end

    for _, value in ipairs({"name", "distance", "item", "team"}) do
        local wanted = (value == "item" and State.heldItem) or (value == "team" and State.teamTag) or (value == "name" and State.name) or (value == "distance" and State.distance)

        if wanted and not bundle[value] then
            bundle[value] = newText(bundle.relationshipColor)
            failed = failed or bundle[value] == nil
        end
    end

    if State.skeleton and #bundle.lines < BONE_COUNT then
        for i = #bundle.lines + 1, BONE_COUNT do
            local line = newDrawing("Line")

            if not line then
                failed = true
                break
            end

            line.Color = bundle.relationshipColor or Color3.fromRGB(170, 70, 255)
            bundle.lines[i] = line
        end
    end

    bundle.nextDrawingRetry = failed and (now + 1) or 0
end

local function drawOffscreenArrow(bundle, camera, screenPos, depth)
    local arrow = ensureArrow(bundle)

    if not arrow then
        return
    end

    local viewportSize = camera.ViewportSize
    local vector2 = Vector2.new(viewportSize.X * 0.5, viewportSize.Y * 0.5)
    local dir = screenPos - vector2

    if depth <= 0 then
        dir = -dir
    end

    if dir.Magnitude < 0.001 then
        dir = Vector2.new(0, -1)
    end

    local unit = dir.Unit
    local radius = math.max(40, math.min(viewportSize.X, viewportSize.Y) * 0.42)
    local tip = vector2 + unit * radius
    local base = tip - unit * 14
    local vector22 = Vector2.new(-unit.Y, unit.X) * 8
    arrow.PointA = tip
    arrow.PointB = base + vector22
    arrow.PointC = base - vector22
    arrow.Visible = true
end

local function refreshBundleRig(bundle, character, humanoid, root)
    if bundle.character == character and bundle.humanoid == humanoid and bundle.root == root then
        return
    end

    bundle.character, bundle.humanoid, bundle.root = character, humanoid, root
    table.clear(bundle.bodyParts)
    table.clear(bundle.skeletonPairs)

    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("BasePart") then
            table.insert(bundle.bodyParts, child)
        end
    end

    local bones = (humanoid.RigType == Enum.HumanoidRigType.R15) and R15_BONES or R6_BONES

    for _, value in ipairs(bones) do
        table.insert(bundle.skeletonPairs, {character:FindFirstChild(value[1]), character:FindFirstChild(value[2])})
    end
end

local function updateBundle(player, bundle, camera, cameraPos, now)
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local humanoidRootPart = character and character:FindFirstChild("HumanoidRootPart")

    if not character or not humanoid or humanoid.Health <= 0 or not humanoidRootPart then
        bundle.character = nil
        bundle.nextVisualUpdate = now + IDLE_INTERVAL
        hideBundle(bundle)
        return
    end

    refreshBundleRig(bundle, character, humanoid, humanoidRootPart)
    local distance = (humanoidRootPart.Position - cameraPos).Magnitude

    if distance > State.maxDistance then
        bundle.nextVisualUpdate = now + IDLE_INTERVAL
        hideBundle(bundle)
        return
    end

    applyRelationshipColor(player, bundle)
    Runtime.prepareESPBundle(bundle, now)

    if bundle.highlight then
        bundle.highlight.Enabled = false
    end

    local rootScreen, onScreen, depth = worldToScreen(camera, humanoidRootPart.Position)

    if not onScreen or depth <= 0 then
        bundle.nextVisualUpdate = now + IDLE_INTERVAL
        hideDrawings(bundle)

        if State.arrow then
            drawOffscreenArrow(bundle, camera, rootScreen, depth)
        end

        return
    end

    local isNear = distance <= NEAR_DISTANCE
    local interval = isNear and NEAR_INTERVAL or FAR_INTERVAL
    bundle.nextVisualUpdate = nextTick(bundle.nextVisualUpdate, now, interval)

    if bundle.arrow then
        bundle.arrow.Visible = false
    end

    local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
    local hasBox = false
    local cornerScreens, cornerDepths = {}, {}

    if State.box or State.name or State.distance or State.heldItem or State.box3d or State.teamTag then
        local boundsMin, boundsMax

        for _, value in ipairs(bundle.bodyParts) do
            if value.Parent == character then
                local halfSize = value.Size * 0.5
                local partMin, partMax = value.Position - halfSize, value.Position + halfSize

                if boundsMin then
                    boundsMin = Vector3.new(math.min(boundsMin.X, partMin.X), math.min(boundsMin.Y, partMin.Y), math.min(boundsMin.Z, partMin.Z))
                    boundsMax = Vector3.new(math.max(boundsMax.X, partMax.X), math.max(boundsMax.Y, partMax.Y), math.max(boundsMax.Z, partMax.Z))
                else
                    boundsMin, boundsMax = partMin, partMax
                end
            end
        end

        if boundsMin then
            local cornerIndex = 0

            for i = 0, 1 do
                for j = 0, 1 do
                    for k = 0, 1 do
                        cornerIndex = cornerIndex + 1
                        local vector = Vector3.new(i == 0 and boundsMin.X or boundsMax.X, j == 0 and boundsMin.Y or boundsMax.Y, k == 0 and boundsMin.Z or boundsMax.Z)
                        local screen, visible, depth2 = worldToScreen(camera, vector)
                        cornerScreens[cornerIndex] = screen
                        cornerDepths[cornerIndex] = depth2

                        if depth2 > 0.5 then
                            hasBox = true
                            minX = math.min(minX, screen.X)
                            minY = math.min(minY, screen.Y)
                            maxX = math.max(maxX, screen.X)
                            maxY = math.max(maxY, screen.Y)
                        end
                    end
                end
            end
        end

        if hasBox then
            local viewportSize = camera.ViewportSize
            minX = math.clamp(minX, -32, viewportSize.X + 32)
            maxX = math.clamp(maxX, -32, viewportSize.X + 32)
            minY = math.clamp(minY, -32, viewportSize.Y + 32)
            maxY = math.clamp(maxY, -32, viewportSize.Y + 32)

            if maxX - minX < 1 or maxY - minY < 1 then
                hasBox = false
            end
        end
    end

    if State.box and bundle.box then
        if hasBox then
            bundle.box.Position = Vector2.new(minX, minY)
            bundle.box.Size = Vector2.new(maxX - minX, maxY - minY)
            bundle.box.Visible = true
        else
            bundle.box.Visible = false
        end
    elseif bundle.box then
        bundle.box.Visible = false
    end

    if State.name and bundle.name then
        if hasBox then
            bundle.name.Text = player.Name .. bundle.relationshipLabel
            bundle.name.Position = Vector2.new((minX + maxX) / 2, minY - 18)
            bundle.name.Visible = true
        else
            bundle.name.Visible = false
        end
    elseif bundle.name then
        bundle.name.Visible = false
    end

    if State.teamTag and bundle.team then
        local teamLabel = hasBox and getTeamLabel(player) or nil

        if teamLabel then
            bundle.team.Text = teamLabel

            if Relationship.Admin(player) and player.Team and player.Team.Name == "STAFF" then
                bundle.team.Color = Color3.fromRGB(255, 200, 50)
            else
                bundle.team.Color = getTeamColor(player)
            end

local function isVisible(camera, part, character)
    local position = camera.CFrame.Position
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {localPlayer.Character, character, camera}
    local raycastResult = workspace:Raycast(position, part.Position - position, raycastParams)
    return raycastResult == nil
end

local function findAimTarget(camera, fov, maxDistance)
    local position = camera.CFrame.Position
    local screenCenter = camera.ViewportSize / 2
    local bestPart, bestDistance = nil, math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and not Relationship.Protected(player) then
            local character = player.Character
            local part = character and getAimPart(character)
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")

            if part and humanoid and humanoid.Health > 0 and (part.Position - position).Magnitude <= maxDistance then
                local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)

                if onScreen and screenPos.Z > 0 then
                    local screenDistance = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude

                    if screenDistance <= fov and screenDistance < bestDistance then
                        if not State.aimbotWallCheck or isVisible(camera, part, character) then
                            bestPart, bestDistance = part, screenDistance
                        end
                    end
                end
            end
        end
    end

    return bestPart
end

        end

            return
        end

        local humanoid = character and character:FindFirstChildOfClass("Humanoid")

        if not humanoid or humanoid.Health <= 0 then
            return
        end

        if State.aimbotMode == "Legit" then
            local tool = character:FindFirstChildOfClass("Tool")

            if not tool then
                return
            end
        end

        local targetPart = findAimTarget(currentCamera, State.aimbotFov, State.aimbotDistance)

        if not targetPart then
            return
        end

        local position = currentCamera.CFrame.Position
        local position2 = targetPart.Position

        if State.aimbotMode == "Rage" then
            currentCamera.CFrame = CFrame.new(position, position2)
        else
            local lookVector = currentCamera.CFrame.LookVector
            local desired = (position2 - position).Unit
            local smoothness = math.clamp(State.aimbotSmoothness, 0.01, 1)
            local alpha = math.clamp(smoothness * 8, 0.02, 0.35)
            local alignment = lookVector:Dot(desired)

            if alignment > 0.998 then
                alpha = 0.6
            end

            local newLook = lookVector:Lerp(desired, alpha)
            currentCamera.CFrame = CFrame.new(position, position + newLook)
        end
    end)
end

    if aimbotConnection then
        aimbotConnection:Disconnect()
        aimbotConnection = nil
    end

do
    local function collectAssemblies(object)
        local roots, seen = {}, {}
        local descendants = object:IsA("BasePart") and {object} or object:GetDescendants()

        for _, value in ipairs(descendants) do
            if value:IsA("BasePart") then
                if value.Anchored then
                    return nil, "O objeto continua fixado pelo servidor."
                end

                local root = value.AssemblyRootPart or value

                if root ~= object and not root:IsDescendantOf(object) then
                    return nil, "O objeto esta ligado a outra estrutura."
                end

                if root.Anchored then
                    return nil, "A estrutura continua fixada pelo servidor."
                end

                if not seen[root] then
                    seen[root] = true
                    roots[#roots + 1] = root
                end
            end
        end

        return roots
    end

    local function canMove(object)
        if not object or not object.Parent or not object:IsDescendantOf(workspace) then
            return false, "Objeto indisponivel."
        end

        local roots, err = collectAssemblies(object)

        if not roots then
            return false, err
        end

        if #roots == 0 then
            return false, "Objeto sem partes fisicas."
        end

        if type(isnetworkowner) ~= "function" then
            return false, "O executor nao informa o controle fisico deste objeto."
        end

        for _, value in ipairs(roots) do
            local success, result = pcall(isnetworkowner, value)

            if not success or result ~= true then
                return false, "O servidor nao concedeu controle fisico deste objeto."
            end
        end

        return true, roots
    end

    Runtime.worldMove = {canMove = canMove}

    function Runtime.worldMove.move(object, cframe, condition)
        local ok, roots = canMove(object)

        if not ok then
            return false, roots
        end

        if Runtime.destroyed or (condition and not condition()) then
            return false, "Acao cancelada."
        end

        for _, value in ipairs(roots) do
            value.AssemblyLinearVelocity, value.AssemblyAngularVelocity = Vector3.zero, Vector3.zero
        end

        if object:IsA("Model") then
            object:PivotTo(cframe)
        elseif object:IsA("BasePart") then
            object.CFrame = cframe
        else
            local handle = object:IsA("Tool") and object:FindFirstChild("Handle")

            if not handle then
                return false, "Objeto sem modelo movivel."
            end

            handle.CFrame = cframe
        end

        task.wait(0.4)

        if not object.Parent or Runtime.destroyed or (condition and not condition()) then
            return false, "Acao interrompida."
        end

        local current = object:IsA("Model") and object:GetPivot() or (object:IsA("BasePart") and object.CFrame) or object:FindFirstChild("Handle").CFrame

        if (current.Position - cframe.Position).Magnitude > 5 then
            return false, "O jogo recusou o movimento do objeto."
        end

        return true
    end

    function Runtime.worldMove.toPosition(object, reference, position, condition)
        local offset = position - reference.Position
        local pivot = object:IsA("Model") and object:GetPivot() or reference.CFrame
        return Runtime.worldMove.move(object, pivot + offset, condition)
    end
end

local function normalizeText(value)
    if not value then
        return ""
    end

    local text = tostring(value)
    text = text:gsub("[áàãâä]", "a"):gsub("[éèêë]", "e"):gsub("[íìîï]", "i")
    text = text:gsub("[óòõôö]", "o"):gsub("[úùûü]", "u"):gsub("ç", "c")
    return text:lower():gsub("[^%w]", "")
end

local normalizeCache, normalizeCacheSize = {}, 0

local function normalize(value)
    if not value then
        return ""
    end

    local key = tostring(value)
    local cached = normalizeCache[key]

    if cached then
        return cached
    end

    local result = normalizeText(key)

    if normalizeCacheSize > 2000 then
        normalizeCache, normalizeCacheSize = {}, 0
    end

    normalizeCache[key] = result
    normalizeCacheSize = normalizeCacheSize + 1
    return result
end

local function matchesWeapon(name)
    local normalized = normalize(name)

    if normalized == "" then
        return false
    end

    if ALLOWED_WEAPONS[normalized] then
        return true
    end

    for key in pairs(ALLOWED_WEAPONS) do
        if (#key >= 4 and normalized:find(key, 1, true)) or (#key <= 3 and normalized:match("^" .. key .. "%d*$")) then
            return true
        end
    end

    return false
end

local weaponCache, weaponCacheSize = {}, 0

local function isAllowedWeapon(name)
    local key = tostring(name or "")
    local cached = weaponCache[key]

    if cached ~= nil then
        return cached
    end

    local result = matchesWeapon(name) and true or false

    if weaponCacheSize > 2000 then
        weaponCache, weaponCacheSize = {}, 0
    end

    weaponCache[key] = result
    weaponCacheSize = weaponCacheSize + 1
    return result
end

                addContainer(folder:FindFirstChild(tostring(localPlayer.UserId)) or folder:FindFirstChild(localPlayer.Name))
            end
        end

        return containers
    end

    local function snapshotInventory()
        local counts = {}

        local function canonicalName(name)
            local normalized = normalize(name)
            return normalized == "ninja" and "fuzilninja" or normalized
        end

        for _, value in pairs({localPlayer.Character, localPlayer:FindFirstChild("Backpack")}) do
            for _, child in ipairs(value:GetChildren()) do
                if child:IsA("Tool") then
                    local weaponName = getWeaponName(child)

                    if weaponName then
                        local key = canonicalName(weaponName)
                        counts[key] = (counts[key] or 0) + 1
                    end
                end
            end
        end

        for _, value in ipairs(getInventoryContainers()) do
            local containerCounts = {}

            local function scan(object)
                local weaponName = getWeaponName(object)

                if weaponName then
                    local key = canonicalName(weaponName)
                    containerCounts[key] = (containerCounts[key] or 0) + readQuantity(object)
                else
                    for _, child in ipairs(object:GetChildren()) do
                        scan(child)
                    end
                end
            end

            for _, child in ipairs(value:GetChildren()) do
                scan(child)
            end

            for key, value2 in pairs(containerCounts) do
                counts[key] = math.max(counts[key] or 0, value2)
            end
        end

        return counts
    end

    local function hasNewWeapon(before, after)
        after = after or snapshotInventory()

        for key, value in pairs(after) do
            if value > (before[key] or 0) then
                return true, key
            end
        end

        return false
    end

    local function getPrimaryPart(object)
        if object:IsA("BasePart") then
            return object
        end

        return (object:IsA("Model") and object.PrimaryPart) or object:FindFirstChildWhichIsA("BasePart", true)
    end

    local dropIndex = setmetatable({}, {__mode = "k"})
    local queued = setmetatable({}, {__mode = "k"})
    local scanQueue, queueHead, queueTail = {}, 1, 0
    local dropFolders, playerCharacters = {}, {}
    local indexReady, forceScanUntil, lastFolderRefresh = false, 0, -math.huge

    local DROP_FOLDER_NAMES = {
        "DropadosInv",
        "Drops",
        "ItensDropados",
        "ItensNoChao",
        "ItemsDrop",
        "DropsInv",
        "ArmasDropadas",

    local function enqueue(object)
        if not object or queued[object] then
            return
        end

        queueTail = queueTail + 1
        scanQueue[queueTail] = object
        queued[object] = true
    end

    local function isOnPlayer(object)
        local character = localPlayer.Character

        if character and (object == character or object:IsDescendantOf(character)) then
            return true
        end

        for _, value in ipairs(playerCharacters) do
            if object == value or object:IsDescendantOf(value) then
                return true
            end
        end

        return false
    end

    local function isInDropArea(object)
        if #dropFolders == 0 then
            return object:IsDescendantOf(workspace)
        end

        for _, value in ipairs(dropFolders) do
            if object == value or object:IsDescendantOf(value) then
                return true
            end
        end

        return false
    end

    local function registerDrop(object, prompt, force)
        if not object or isOnPlayer(object) or not isInDropArea(object) then
            return
        end

        if not (object:IsA("Model") or object:IsA("Tool") or object:IsA("BasePart")) then
            return
        end

        local weaponName = getWeaponName(object)

        if not weaponName and not force then
            return
        end

        local part = getPrimaryPart(object)

        if not part then
            return
        end

        local parent = object.Parent

        for _ = 1, 8 do
            if not parent or parent == workspace then
                break
            end

            if dropIndex[parent] then
                if prompt then
                    dropIndex[parent].prompt = prompt
                end

                return
            end

            parent = parent.Parent
        end

        local existing = dropIndex[object]

        dropIndex[object] = {
            item = object,
            parte = part,
            prompt = prompt or (existing and existing.prompt),
            nome = weaponName or (prompt and prompt.ObjectText ~= "" and prompt.ObjectText) or object.Name,
        }
    end

    local function inspectObject(object)
        if isOnPlayer(object) or not isInDropArea(object) then
            return
        end

        if object:IsA("ProximityPrompt") then
            if not object.Enabled or object.MaxActivationDistance <= 0 then
                return
            end

            local parent = object.Parent

            if parent and parent:IsA("Attachment") then
                parent = parent.Parent
            end

            if not parent then
                return
            end

            local target, node = parent, parent

            for _ = 1, 10 do
                if not node or node == workspace or node:IsA("Folder") then
                    break
                end

                if (node:IsA("Model") or node:IsA("Tool")) and getWeaponName(node) then
                    target = node
                    break
                end

                node = node.Parent
            end

            local promptText = (object.ActionText .. " " .. object.ObjectText):lower()
            local looksLikeWeapon = isWeaponName(object.ObjectText) or promptText:find("%f[%w]arma%f[%W]") or promptText:find("%f[%w]weapon%f[%W]")

            if target == parent and looksLikeWeapon then
                local model = parent:FindFirstAncestorWhichIsA("Model")

                if model and model.Parent and not isOnPlayer(model) and isInDropArea(model) then
                    local useModel = false

                    for _, value in ipairs(dropFolders) do
                        if model.Parent == value then
                            useModel = true
                            break
                        end
                    end

                    if not useModel and #dropFolders == 0 then
                        local children = model:GetChildren()

                        if #children <= 24 and not model:FindFirstChildOfClass("Humanoid") then
                            useModel = true

                            for _, value in ipairs(children) do
                                if value:IsA("Folder") or value:IsA("Model") then
                                    useModel = false
                                    break
                                end
                            end
                        end
                    end

                    if useModel then
                        target = model
                    end
                end
            end

            registerDrop(target, object, looksLikeWeapon)
        elseif object:IsA("Model") or object:IsA("Tool") or object:IsA("BasePart") then
            registerDrop(object)
        end
    end

    local function refreshDropFolders()
        playerCharacters = {}

        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then
                playerCharacters[#playerCharacters + 1] = player.Character
            end
        end

        local folders = {}

        for _, value in ipairs(DROP_FOLDER_NAMES) do
            local folder = workspace:FindFirstChild(value)

            if folder then
                folders[#folders + 1] = folder

                if not table.find(dropFolders, folder) then
                    enqueue(folder)
                end
            end
        end

        local scanWorkspace = #folders == 0 and (not indexReady or #dropFolders > 0)
        dropFolders, indexReady, lastFolderRefresh = folders, true, os.clock()

        if scanWorkspace then
            enqueue(workspace)
        end
    end

    trackConnection(workspace.DescendantAdded:Connect(function(descendant)
        if not indexReady or not isInDropArea(descendant) or isOnPlayer(descendant) then
            return
        end

        enqueue(descendant)
        local parent = descendant.Parent

        for _ = 1, 4 do
            if not parent or parent == workspace or parent:IsA("Folder") then
                break
            end

            enqueue(parent)
            parent = parent.Parent
        end
    end))

    trackConnection(workspace.ChildAdded:Connect(function(child)
        if table.find(DROP_FOLDER_NAMES, child.Name) then
            lastFolderRefresh = -math.huge
        end
    end))

    task.spawn(function()
        while collectorActive and not Runtime.destroyed do
            if State.autoColetaEnabled or State.coletaHudEnabled or os.clock() < forceScanUntil then
                local success, result = pcall(function()
                    if os.clock() - lastFolderRefresh >= 3 then
                        refreshDropFolders()
                    end

                    local processed, budgetEnd = 0, os.clock() + 0.003

                    while queueHead <= queueTail and processed < 80 and os.clock() < budgetEnd do
                        local object = scanQueue[queueHead]
                        scanQueue[queueHead] = nil
                        queueHead = queueHead + 1
                        queued[object] = nil
                        processed = processed + 1

                        if object and (object == workspace or object.Parent) and not isOnPlayer(object) then
                            if object ~= workspace then
                                inspectObject(object)
                            end

                            if object == workspace or object:IsA("Folder") or object:IsA("Model") or object:IsA("Tool") or object:IsA("BasePart") or object:IsA("Attachment") then
                                local children = object:GetChildren()

                                for index, value in ipairs(children) do
                                    enqueue(value)

                                    if index % 120 == 0 then
                                        task.wait()

                                        if not collectorActive or Runtime.destroyed then
                                            return
                                        end
                                    end
                                end
                            end
                        end
                    end

                    if queueHead > queueTail then
                        scanQueue, queueHead, queueTail = {}, 1, 0
                    end

                    Runtime.dropIndexBusy = queueHead <= queueTail
                end)

                if not success then
                    setColetaStatus("Falha ao indexar drops: " .. tostring(result))
                end

                task.wait(Runtime.dropIndexBusy and 0.03 or 0.15)
            else
                task.wait(0.2)
            end
        end
    end)

    local function listDrops()
        forceScanUntil = os.clock() + 3

        if not indexReady then
            Runtime.dropIndexBusy = true
        end

        playerCharacters = {}

        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then
                playerCharacters[#playerCharacters + 1] = player.Character
            end
        end

        local rootPart, drops = getRootPart(), {}

        if not rootPart then
            return drops
        end

        for key, value in pairs(dropIndex) do
            if not key.Parent or not value.parte.Parent or not key:IsDescendantOf(workspace) or isOnPlayer(key) or not isInDropArea(key) then
                dropIndex[key] = nil
            else
                local nested, parent = false, key.Parent

                for _ = 1, 8 do
                    if not parent or parent == workspace then
                        break
                    end

                    if dropIndex[parent] then
                        nested = true
                        break
                    end

                    parent = parent.Parent
                end

                if not nested then
                    value.distancia = (rootPart.Position - value.parte.Position).Magnitude
                    drops[#drops + 1] = value
                end
            end
        end

        table.sort(drops, function(a, b)
            return a.distancia < b.distancia
        end)

        return drops
    end

        setColetaStatus("Arma recebida; retornando ao hospital...")

        if not ok then
            setColetaStatus("Falha no retorno: " .. (err or "TP recusado"))
        else
            setColetaStatus("Retorno concluido")
        end

        return ok
    end

    local function interactWithDrop(drop, condition, skipFire)
        local prompt = drop.prompt

        if not prompt or not prompt.Parent then
            prompt = drop.item:FindFirstChildWhichIsA("ProximityPrompt", true)
        end

        if prompt and prompt.Enabled then
            return Runtime.interact.prompt(prompt, condition, skipFire)
        end

        local clickDetector = drop.item:FindFirstChildWhichIsA("ClickDetector", true)

        if clickDetector and type(fireclickdetector) == "function" then
            return pcall(fireclickdetector, clickDetector)
        end

        local touchPart = drop.item:FindFirstChild("Handle", true) or drop.parte
        local rootPart = getRootPart()

        if rootPart and touchPart and type(firetouchinterest) == "function" then
            pcall(firetouchinterest, rootPart, touchPart, 0)
            task.wait(0.1)
            return pcall(firetouchinterest, rootPart, touchPart, 1)
        end

        return false
    end

    collectItem = function(drop, confirm, pull)
        if Runtime.collectionBusy then
            return false, "Aguarde a acao atual."
        end

        if not drop or not drop.item or not drop.parte or not drop.item.Parent or not drop.parte.Parent then
            return false, "Item indisponivel."
        end

        local success, result = pcall(snapshotInventory)

        if not success then
            return false, "Inventario indisponivel: " .. tostring(result)
        end

        Runtime.collectionBusy = true
        Runtime.itemTeleportRunId = Runtime.itemTeleportRunId + 1
        local itemTeleportRunId, character = Runtime.itemTeleportRunId, localPlayer.Character
        local interacted = false

        local function isCollected()
            if confirm then
                return confirm()
            end

            return hasNewWeapon(result) or (interacted and not drop.item.Parent)
        end

        local success2, result2, result22 = pcall(function()
            setColetaStatus("Coletando " .. tostring(drop.nome or drop.item.Name))

            if not confirm and (pull == true or (pull == nil and State.pullWeaponsEnabled)) then
                local rootPart = getRootPart()

                if not rootPart then
                    return false, "Personagem indisponivel."
                end

                local targetPos = rootPart.Position + rootPart.CFrame.LookVector * 3 + Vector3.new(0, -1, 0)
                local moved, err = Runtime.worldMove.toPosition(drop.item, drop.parte, targetPos, nil)

                if not moved then
                    return false, err
                end
            else
                local rootPart = getRootPart()

                if not rootPart then
                    return false, "Personagem indisponivel."
                end

                local proximityPrompt = drop.prompt or drop.item:FindFirstChildWhichIsA("ProximityPrompt", true)
                local promptPos = proximityPrompt and Runtime.interact.position(proximityPrompt) or drop.parte.Position

                if not promptPos then
                    return false, "Posicao da arma indisponivel."
                end

                local heightOffset = drop.parte.Size.Y * 0.5 + 2.5

                if proximityPrompt then
                    heightOffset = math.min(heightOffset, proximityPrompt.MaxActivationDistance * 0.5)
                end

                local targetPos = promptPos + Vector3.new(0, heightOffset, 0)
                local teleported = Runtime.teleportLocation(targetPos, "Arma")

                if not teleported then
                    return false, "TP para a arma falhou."
                end

                Runtime.fixedTeleportBusy = false
                Runtime.travelBusy = false
                task.wait(0.25)
            end

            local rootPart = getRootPart()

            if not rootPart then
                return false, "Personagem indisponivel."
            end

            local proximityPrompt = drop.prompt or drop.item:FindFirstChildWhichIsA("ProximityPrompt", true)
            local clickDetector = drop.item:FindFirstChildWhichIsA("ClickDetector", true)
            local touchPart = drop.item:FindFirstChild("Handle", true) or drop.parte

            for _ = 1, 4 do
                if isCollected() or not drop.item.Parent then
                    break
                end

                local fired = false

                if proximityPrompt and proximityPrompt.Parent and proximityPrompt.Enabled then
                    if type(fireproximityprompt) == "function" then
                        if pcall(fireproximityprompt, proximityPrompt, proximityPrompt.HoldDuration or 0) then
                            fired = true
                        end
                    end

                    if not fired then
                        if pcall(function()
                            proximityPrompt:InputHoldBegin()
                            task.wait(math.max(0.05, proximityPrompt.HoldDuration or 0))
                            proximityPrompt:InputHoldEnd()
                        end) then
                            fired = true
                        end
                    end
                end

                if not fired and clickDetector and type(fireclickdetector) == "function" then
                    if pcall(fireclickdetector, clickDetector) then
                        fired = true
                    end
                end

                if not fired and type(firetouchinterest) == "function" and rootPart and touchPart then
                    pcall(firetouchinterest, rootPart, touchPart, 0)
                    task.wait(0.08)
                    pcall(firetouchinterest, rootPart, touchPart, 1)
                    fired = true
                end

                if fired then
                    interacted = true
                    local deadline = os.clock() + 0.5

                    while os.clock() < deadline do
                        if isCollected() then
                            break
                        end

                        if not drop.item.Parent then
                            interacted = true
                            break
                        end

                        task.wait(0.05)
                    end
                end

                task.wait(0.15)
            end

            if not isCollected() then
                return false, "O jogo nao confirmou a arma nos dados do inventario."
            end

            lastInventorySnapshot = snapshotInventory()

            if not confirm and State.returnOnWeaponPickup ~= false then
                setColetaStatus("Arma coletada! Retornando ao hospital...")
                task.wait(0.3)
                local returned = Runtime.teleportLocation(HOSPITAL_POSITION, "Hospital")

                if not returned then
                    return false, "Falha ao retornar ao hospital."
                end

                Runtime.fixedTeleportBusy = false
                Runtime.travelBusy = false
            end

            return true, "Arma coletada com sucesso."
        end)

        Runtime.collectionBusy = false

        if not success2 then
            setColetaStatus("Falha na coleta: " .. tostring(result2))
            return false, Runtime.coletaStatus
        end

        if not result2 then
            setColetaStatus(result22 or "Coleta interrompida")
        end

        return result2, result22
    end

    function Runtime.stopAutoColeta()
        State.autoColetaEnabled = false
        Runtime.coletaRunId, Runtime.itemTeleportRunId = Runtime.coletaRunId + 1, Runtime.itemTeleportRunId + 1
        Runtime.travel.cancel("collection")

        if Runtime.coletaControl then
            Runtime.coletaControl:Set(false, true)
        end
    end

    function Runtime.startAutoColeta()
        if State.autoFarmPumpkinEnabled and Runtime.startPumpkin then
            Runtime.startPumpkin(false)
        end

        if State.autoFarmLixeiroEnabled and Runtime.startLixeiro then
            Runtime.startLixeiro(false)
        end

        if Runtime.collectionWorker and Runtime.collectionWorker.running and State.autoColetaEnabled then
            return
        end

        if Runtime.startPesca and State.autoFarmPescaEnabled then
            Runtime.startPesca(false)
        end

        State.autoColetaEnabled, Runtime.coletaRunId = true, Runtime.coletaRunId + 1
        local coletaRunId, worker = Runtime.coletaRunId, {running = true}
        Runtime.collectionWorker = worker

        if Runtime.coletaControl then
            Runtime.coletaControl:Set(true, true)
        end

        task.spawn(function()
            local success, result = pcall(function()
                local cooldowns = setmetatable({}, {__mode = "k"})

                while collectorActive and State.autoColetaEnabled and not Runtime.destroyed and Runtime.coletaRunId == coletaRunId do
                    if not Runtime.collectionBusy and not Runtime.travel.owner then
                        local target

                        for _, value in ipairs(listDrops()) do
                            if (cooldowns[value.item] or 0) <= os.clock() then
                                target = value
                                break
                            end
                        end

                        if target then
                            local ok, err = collectItem(target)

                            if not ok then
                                cooldowns[target.item] = os.clock() + 4
                                setColetaStatus(err or "Coleta recusada")
                            end
                        else
                            setColetaStatus("Aguardando arma dropada carregada no mapa")
                        end
                    end

                    task.wait(math.max(0.25, tonumber(State.autoColetaCooldown) or 0.35))
                end
            end)

            worker.running = false

            if not success and Runtime.coletaRunId == coletaRunId then
                State.autoColetaEnabled = false

                if Runtime.coletaControl then
                    Runtime.coletaControl:Set(false, true)
                end

                setColetaStatus("Erro na Auto Coleta: " .. tostring(result))
                notify(Runtime.coletaStatus)
            end
        end)
    end

    local function handleWeaponReceived()
        if Runtime.collectionBusy or Runtime.destroyed or State.returnOnWeaponPickup == false then
            return
        end

        Runtime.collectionBusy = true

        if State.autoFarmPumpkinEnabled and Runtime.startPumpkin then
            Runtime.startPumpkin(false)
        end

        if State.autoFarmLixeiroEnabled and Runtime.startLixeiro then
            Runtime.startLixeiro(false)
        end

        if State.autoFarmPescaEnabled and Runtime.startPesca then
            Runtime.startPesca(false)
        end

        if State.farmPartesEnabled and Runtime.pararFarmPartes then
            Runtime.pararFarmPartes()
        end

        Runtime.travel.cancel()
        Runtime.itemTeleportRunId = Runtime.itemTeleportRunId + 1
        local itemTeleportRunId, character = Runtime.itemTeleportRunId, localPlayer.Character

        task.spawn(function()
            local function stillValid()
                return collectorActive and not Runtime.destroyed and State.returnOnWeaponPickup ~= false and Runtime.itemTeleportRunId == itemTeleportRunId and localPlayer.Character == character
            end

            local deadline = os.clock() + 3

            while Runtime.travel.owner and stillValid() and os.clock() < deadline do
                task.wait(0.05)
            end

            local success, result = pcall(function()
                if stillValid() then
                    returnToHospital(stillValid)
                end
            end)

            Runtime.collectionBusy = false

            if not success then
                setColetaStatus("Erro no retorno: " .. tostring(result))
            end
        end)
    end

    task.spawn(function()
        while collectorActive and not Runtime.destroyed do
            task.wait(INVENTORY_POLL_INTERVAL)
            local success, result = pcall(snapshotInventory)

            if success then
                local previous = lastInventorySnapshot
                lastInventorySnapshot = result

                if previous and os.clock() >= snapshotReadyAt and hasNewWeapon(previous, result) and not Runtime.collectionBusy then
                    handleWeaponReceived()
                end
            end
        end
    end)

    trackConnection(localPlayer.CharacterAdded:Connect(function()
        lastInventorySnapshot, snapshotReadyAt = nil, os.clock() + 0.75
    end))

    lastInventorySnapshot = snapshotInventory()

    function Runtime.cleanupCollection()
        collectorActive = false
        Runtime.stopAutoColeta()
    end
end

do
    local JOB_SWITCH_TIMEOUT = 10
    local LIXEIRO_POSITION = Vector3.new(-1216.14, 129.42, -5183.68)

    local function getTeamName()
        return localPlayer.Team and localPlayer.Team.Name or nil
    end

    local function getLevel()
        local newStats = localPlayer:FindFirstChild("NewStats")
        local level = newStats and newStats:FindFirstChild("Level")
        return level and tonumber(level.Value) or 0
    end

    local function getJobs()
        local success, result = pcall(function()
            return require(ReplicatedStorage.Mercadinho.PrefModule)
        end)

        if not success or type(result) ~= "table" or type(result.Empregos) ~= "table" then
            return {}
        end

        local jobs = {}

        for key, value in pairs(result.Empregos) do
            table.insert(jobs, {nome = key, level = tonumber(value.Level) or 0, salario = tonumber(value.Salario) or 0})
        end

        table.sort(jobs, function(a, b)
            if a.level == b.level then
                return a.nome < b.nome
            end

            return a.level < b.level
        end)

        return jobs
    end

    local function countGarbageBags()
        local inv = localPlayer:FindFirstChild("Inv")
        local count = 0

        if inv then
            for _, child in ipairs(inv:GetChildren()) do
                if child.Name == "Saco de Lixo" then
                    count = count + 1
                end
            end
        end

        return count
    end

    local function getMoney()
        local leaderstats = localPlayer:FindFirstChild("leaderstats")
        local dinheiro = leaderstats and leaderstats:FindFirstChild("Dinheiro")
        return dinheiro and tonumber(dinheiro.Value) or nil
    end

    local function getLixeiroJob()
        for _, value in ipairs(getJobs()) do
            if string.lower(value.nome) == "lixeiro" then
                return value
            end
        end

        return {nome = "Lixeiro", level = 1, salario = 0}
    end

    local function setGarbageStatus(status)
        Runtime.garbageFarmStatus = tostring(status or "")
    end

    local function garbageActive(runId)
        return State.autoFarmLixeiroEnabled and not Runtime.destroyed and Runtime.garbageFarmRunId == runId
    end

    local function waitWhileActive(seconds, runId)
        local deadline = os.clock() + math.max(0, seconds or 0)

        repeat
            if not garbageActive(runId) then
                return false
            end

            task.wait(math.min(0.1, math.max(0.01, deadline - os.clock())))
        until os.clock() >= deadline

        return garbageActive(runId)
    end

    local function confirmJob(remote, job)
        remote:FireServer(job.nome, job.level, true)
        local playerGui = localPlayer:FindFirstChild("PlayerGui")
        local lixeiroUI = playerGui and playerGui:FindFirstChild("LixeiroUI")
        local lixeiroFrame = lixeiroUI and lixeiroUI:FindFirstChild("LixeiroFrame")

        if lixeiroFrame and lixeiroFrame.Visible then
            lixeiroFrame.Visible = false
        end
    end

    local function ensureLixeiroJob(runId)
        if getTeamName() == "Lixeiro" then
            return true
        end

        local mercadinho = ReplicatedStorage:FindFirstChild("Mercadinho")
        local prefRemote = mercadinho and mercadinho:FindFirstChild("PrefRemote")

        if not prefRemote or not prefRemote:IsA("RemoteEvent") then
            setGarbageStatus("PrefRemote nao encontrado")
            return false
        end

        local job = getLixeiroJob()

        if getLevel() < job.level then
            return false
        end

        setGarbageStatus("Selecionando lixeiro...")
        prefRemote:FireServer(job.nome, job.level)
        waitWhileActive(0.25, runId)

        if not garbageActive(runId) then
            return false
        end

        confirmJob(prefRemote, job)
        local deadline = os.clock() + JOB_SWITCH_TIMEOUT

        repeat
            if not garbageActive(runId) then
                return false
            end

            if getTeamName() == "Lixeiro" then
                waitWhileActive(1.2, runId)
                return true
            end

            task.wait(0.15)
        until os.clock() >= deadline

        return false
    end

    local function isCollectableTrash(part, prompt)
        return part and part.Parent and part:IsA("BasePart") and part.Name == "Lixo" and part.Transparency < 0.95 and (prompt and prompt.Parent == part and prompt:IsA("ProximityPrompt") and prompt.Enabled)
    end

    local function findTrash()
        local rootPart = getRootPart()
        local lixeiro = workspace:FindFirstChild("Lixeiro")
        local trash = {}

        if not rootPart or not lixeiro then
            return trash
        end

        for _, child in ipairs(lixeiro:GetChildren()) do
            local pegarLixo = child:FindFirstChild("PegarLixo")

            if isCollectableTrash(child, pegarLixo) then
                table.insert(trash, {
                    parte = child,
                    prompt = pegarLixo,
                    posicao = child.Position,
                    distancia = (child.Position - rootPart.Position).Magnitude,
                })
            end
        end

        table.sort(trash, function(a, b)
            return a.distancia < b.distancia
        end)

        return trash
    end

    local function isInRange(rootPart, part, prompt)
        if not rootPart or not part or not part.Parent then
            return false
        end

        local range = prompt and math.max(2.5, prompt.MaxActivationDistance - 0.65) or 4.25
        return (rootPart.Position - part.Position).Magnitude <= range
    end

    local function moveToTarget(part, prompt, runId)
        local rootPart = getRootPart()

        if not rootPart or not part or not part.Parent or not garbageActive(runId) then
            return false
        end

        if isInRange(rootPart, part, prompt) then
            return true
        end

        local promptPos = prompt and Runtime.interact.position(prompt) or part.Position

        if not promptPos then
            return false
        end

        local heightOffset = math.min(part.Size.Y * 0.5 + 2.5, prompt and prompt.MaxActivationDistance * 0.6 or 3)

        local ok, err = Runtime.travel.to("garbage", promptPos + Vector3.new(0, heightOffset, 0), function()
            return garbageActive(runId) and part.Parent ~= nil and (not prompt or prompt.Parent ~= nil)
        end)

        if not ok then
            setGarbageStatus(err or "TP interrompido")
            return false
        end

        return isInRange(getRootPart(), part, prompt)
    end

    local function firePrompt(prompt, holdManually)
        if not prompt or not prompt.Parent or not prompt.Enabled then
            return false
        end

        if not holdManually and type(fireproximityprompt) == "function" then
            return pcall(fireproximityprompt, prompt, 0)
        end

        return pcall(function()
            prompt:InputHoldBegin()
            task.wait(math.max(0.15, prompt.HoldDuration + 0.1))
            prompt:InputHoldEnd()
        end)
    end

    local function pickUpTrash(part, prompt, runId)
        local bagsBefore = countGarbageBags()

        for i = 1, 3 do
            if not garbageActive(runId) or not isCollectableTrash(part, prompt) then
                return not part.Parent or part.Transparency >= 0.95
            end

            local rootPart = getRootPart()

            if not isInRange(rootPart, part, prompt) then
                return false
            end

            firePrompt(prompt, i > 1)
            local deadline = os.clock() + 2.75

            repeat
                if not garbageActive(runId) then
                    return false
                end

                if not part.Parent or part.Transparency >= 0.95 or countGarbageBags() > bagsBefore then
                    return true
                end

                task.wait(0.1)
            until os.clock() >= deadline
        end

        return false
    end

    local function getDropOff()
        local entregaLixo = workspace:FindFirstChild("EntregaLixo")
        local lixeira = entregaLixo and entregaLixo:FindFirstChild("Lixeira")
        local entregarLixo = lixeira and lixeira:FindFirstChild("EntregarLixo")

        if lixeira and lixeira:IsA("BasePart") and entregarLixo and entregarLixo:IsA("ProximityPrompt") then
            return lixeira, entregarLixo
        end

        return nil, nil
    end

    local function deliverTrash(count, runId)
        local bin, prompt = getDropOff()

        if not bin or not prompt then
            return false
        end

        setGarbageStatus(string.format("%d/%d coletados; indo entregar...", count, count))

        if not moveToTarget(bin, prompt, runId) then
            if not garbageActive(runId) then
                return false
            end

            return false
        end

        local bagsBefore = countGarbageBags()
        local moneyBefore = getMoney()

        for i = 1, 3 do
            if not garbageActive(runId) then
                return false
            end

            firePrompt(prompt, i > 1)
            local deadline = os.clock() + 4

            repeat
                if not garbageActive(runId) then
                    return false
                end

                local bagsNow = countGarbageBags()
                local moneyNow = getMoney()

                if (bagsBefore > 0 and bagsNow < bagsBefore) or (moneyBefore and moneyNow and moneyNow > moneyBefore) then
                    return true
                end

                task.wait(0.15)
            until os.clock() >= deadline
        end

        return false
    end

    function Runtime.teleportLixeiro(condition)
        return Runtime.travel.to("garbage", LIXEIRO_POSITION, condition)
    end

    local function runGarbageFarm(runId)
        local deadline = os.clock() + 3

        while Runtime.travel.owner and garbageActive(runId) and os.clock() < deadline do
            task.wait(0.05)
        end

        if not garbageActive(runId) then
            return
        end

        setGarbageStatus("TP para o ponto do lixeiro...")

        local ok, err = Runtime.teleportLixeiro(function()
            return garbageActive(runId)
        end)

        if not ok then
            setGarbageStatus(err or "TP recusado")
            State.autoFarmLixeiroEnabled = false

            if Runtime.lixeiroControl then
                Runtime.lixeiroControl:Set(false, true)
            end

            notify(Runtime.garbageFarmStatus)
            return
        end

        while garbageActive(runId) do
            if not ensureLixeiroJob(runId) then
                waitWhileActive(2, runId)
            else
                local target = math.clamp(math.floor(tonumber(State.autoFarmLixeiroQuantidade) or 20), 1, 20)
                local collected = 0

                while garbageActive(runId) and getTeamName() == "Lixeiro" and collected < target do
                    local trash = findTrash()

                    if #trash == 0 then
                        setGarbageStatus(string.format("Coletados %d/%d; aguardando...", collected, target))
                        waitWhileActive(1.25, runId)
                    else
                        local nearest = trash[1]

                        if moveToTarget(nearest.parte, nearest.prompt, runId) then
                            if pickUpTrash(nearest.parte, nearest.prompt, runId) then
                                collected = collected + 1
                                waitWhileActive(0.45, runId)
                            else
                                if not garbageActive(runId) then
                                    break
                                end

                                waitWhileActive(0.35, runId)
                            end
                        else
                            if not garbageActive(runId) then
                                break
                            end

                            waitWhileActive(0.4, runId)
                        end
                    end
                end

                if not garbageActive(runId) then
                    break
                end

                if getTeamName() ~= "Lixeiro" then
                    setGarbageStatus("Emprego alterado; retomando...")
                elseif collected >= target then
                    while garbageActive(runId) and not deliverTrash(collected, runId) do
                        waitWhileActive(1, runId)
                    end

                    if garbageActive(runId) then
                        waitWhileActive(1.25, runId)
                    end
                end
            end
        end
    end

    local function setLixeiroFarm(enabled)
        State.autoFarmLixeiroEnabled = enabled == true
        Runtime.garbageFarmRunId = Runtime.garbageFarmRunId + 1
        local garbageFarmRunId = Runtime.garbageFarmRunId
        Runtime.travel.cancel("garbage")

        if State.autoFarmLixeiroEnabled then
            if State.autoFarmPumpkinEnabled and Runtime.startPumpkin then
                Runtime.startPumpkin(false)
            end

            if Runtime.stopAutoColeta then
                Runtime.stopAutoColeta()
            end

            if State.autoFarmPescaEnabled and Runtime.startPesca then
                Runtime.startPesca(false)
            end

            if State.farmPartesEnabled and Runtime.pararFarmPartes then
                Runtime.pararFarmPartes()
            end

            setGarbageStatus("Verificando emprego...")

            task.spawn(function()
                local success, result = pcall(runGarbageFarm, garbageFarmRunId)

                if not success and garbageActive(garbageFarmRunId) then
                    setGarbageStatus("Erro: " .. tostring(result))
                    State.autoFarmLixeiroEnabled = false
                    Runtime.garbageFarmRunId = Runtime.garbageFarmRunId + 1

                    if Runtime.lixeiroControl then
                        Runtime.lixeiroControl:Set(false, true)
                    end
                end
            end)
        else
            setGarbageStatus("Auto Farm parado.")
            Runtime.travel.cancel("garbage")
        end
    end

    Runtime.startLixeiro = setLixeiroFarm
end

do
    local FISHING_POSITION = Vector3.new(-2593.61, 122.34, -3044.99)

    local function simplify(text)
        return tostring(text or ""):lower():gsub("[^%w]", "")
    end

    local function fishingActive(runId)
        return State.autoFarmPescaEnabled and not Runtime.destroyed and Runtime.pescaRunId == runId
    end

    local function isFishingPrompt(prompt)
        local text = simplify(prompt.Name .. " " .. prompt.ActionText .. " " .. prompt.ObjectText)
        return text:find("pesc", 1, true) or text:find("fishing", 1, true) or text:find("fish", 1, true)
    end

    local function findFishingRod()
        local containers = {localPlayer.Character, localPlayer:FindFirstChild("Backpack")}

        for _, value in pairs(containers) do
            for _, child in ipairs(value:GetChildren()) do
                if child:IsA("Tool") then
                    local name = simplify(child.Name)

                    if name:find("varadepesca", 1, true) or name == "vara" or name:find("fishingrod", 1, true) or name:find("fishingpole", 1, true) then
                        return child
                    end
                end
            end
        end
    end

    local function runFishingFarm(runId)
        while fishingActive(runId) do
            local character, rootPart = localPlayer.Character, getRootPart()

            if not rootPart then
                task.wait(0.3)
            else
                Runtime.pescaStatus = "Indo ao ponto de pesca"

                local ok, err = Runtime.travel.to("fishing", FISHING_POSITION, function()
                    return fishingActive(runId)
                end)

                if not ok then
                    Runtime.pescaStatus = err or "Deslocamento interrompido"

                    if fishingActive(runId) then
                        notify(Runtime.pescaStatus)
                    end

                    break
                end

                local prompts, nextScan, nextCast = {}, 0, 0
                local scanning = false

                while fishingActive(runId) and localPlayer.Character == character do
                    rootPart = getRootPart()

                    if not rootPart or (rootPart.Position - FISHING_POSITION).Magnitude > 8 then
                        Runtime.pescaStatus = "Voce saiu do ponto; farm pausado"
                        break
                    end

                    if os.clock() >= nextScan and not scanning then
                        nextScan, scanning = os.clock() + 2, true

                        task.spawn(function()
                            local success, result = pcall(function()
                                local found, queue, head, tail = {}, {workspace}, 1, 1

                                while head <= tail and fishingActive(runId) and localPlayer.Character == character do
                                    local node = queue[head]
                                    queue[head], head = nil, head + 1

                                    for _, child in ipairs(node:GetChildren()) do
                                        tail = tail + 1
                                        queue[tail] = child

                                        if child:IsA("ProximityPrompt") and child.Enabled and isFishingPrompt(child) then
                                            local promptPos, rootPart2 = Runtime.interact.position(child), getRootPart()

                                            if promptPos and rootPart2 and (promptPos - rootPart2.Position).Magnitude <= child.MaxActivationDistance then
                                                found[#found + 1] = child
                                            end
                                        end
                                    end

                                    if head % 100 == 0 then
                                        task.wait()
                                    end
                                end

                                return found
                            end)

                            if success and fishingActive(runId) and localPlayer.Character == character then
                                prompts = result
                            end

                            scanning = false
                        end)
                    end

                    local acted = false

                    if not Runtime.actionsBusy then
                        for _, value in ipairs(prompts) do
                            if not fishingActive(runId) then
                                break
                            end

                            if value.Parent then
                                acted = Runtime.interact.prompt(value, function()
                                    return fishingActive(runId)
                                end) or acted
                            end
                        end

                        if not acted and os.clock() >= nextCast then
                            local rod, humanoid = findFishingRod(), getHumanoid()

                            if rod and humanoid and rod.Enabled then
                                if rod.Parent ~= character then
                                    humanoid:EquipTool(rod)
                                    task.wait(0.15)
                                end

                                if fishingActive(runId) and rod.Parent == character then
                                    rod:Activate()
                                    nextCast, acted = os.clock() + 1.5, true
                                end
                            end
                        end
                    end

                    Runtime.pescaStatus = acted and "Pescando" or "Aguardando vara ou interacao de pesca"
                    task.wait(0.35)
                end

                if localPlayer.Character == character then
                    break
                end
            end
        end

        if Runtime.pescaRunId == runId then
            State.autoFarmPescaEnabled = false

            if Runtime.pescaControl then
                Runtime.pescaControl:Set(false, true)
            end
        end
    end

    function Runtime.startPesca(enabled)
        Runtime.pescaRunId = Runtime.pescaRunId + 1
        State.autoFarmPescaEnabled = enabled == true
        Runtime.travel.cancel("fishing")

        if Runtime.pescaControl then
            Runtime.pescaControl:Set(State.autoFarmPescaEnabled, true)
        end

        if not State.autoFarmPescaEnabled then
            Runtime.pescaStatus = "Parado"
            return
        end

        if State.autoFarmPumpkinEnabled and Runtime.startPumpkin then
            Runtime.startPumpkin(false)
        end

        if Runtime.stopAutoColeta then
            Runtime.stopAutoColeta()
        end

        if State.autoFarmLixeiroEnabled and Runtime.startLixeiro then
            Runtime.startLixeiro(false)
        end

        local pescaRunId = Runtime.pescaRunId

        task.spawn(function()
            while Runtime.travel.owner and fishingActive(pescaRunId) do
                task.wait(0.05)
            end

            local success, result = pcall(runFishingFarm, pescaRunId)

            if not success and Runtime.pescaRunId == pescaRunId then
                State.autoFarmPescaEnabled, Runtime.pescaStatus = false, "Falha: " .. tostring(result)

                if Runtime.pescaControl then
                    Runtime.pescaControl:Set(false, true)
                end

                notify(Runtime.pescaStatus)
            end
        end)
    end
end

do
    local TRACER_COLORS = {
        {"Ciano", Color3.fromRGB(0, 255, 255)},
        {"Verde", Color3.fromRGB(60, 255, 90)},
        {"Rosa", Color3.fromRGB(255, 60, 190)},
        {"Roxo", Color3.fromRGB(160, 70, 255)},
        {"Vermelho", Color3.fromRGB(255, 45, 45)},
        {"Azul", Color3.fromRGB(70, 130, 255)},
        {"Laranja", Color3.fromRGB(255, 145, 30)},
        {"Branco", Color3.fromRGB(240, 240, 240)},
        {"Amarelo", Color3.fromRGB(255, 230, 60)},
        {"VerdeLimao", Color3.fromRGB(180, 255, 0)},
        {"Magenta", Color3.fromRGB(255, 0, 200)},
        {"Turquesa", Color3.fromRGB(0, 200, 180)},
        {"Arco-iris", nil},
    }

    local originalBulletColor = nil

    local function getBulletColorValue()
        local orpheusGunCFG = localPlayer:FindFirstChild("OrpheusGunCFG")
        local corBala = orpheusGunCFG and orpheusGunCFG:FindFirstChild("CorBala")

        if corBala and originalBulletColor == nil then
            originalBulletColor = corBala.Color
        end

        return corBala
    end

    local function applyTracerColor()
        local colorValue = getBulletColorValue()

        if not colorValue then
            return
        end

        if not State.armaTracer then
            if originalBulletColor then
                pcall(function()
                    colorValue.Color = originalBulletColor
                end)
            end

            return
        end

        if State.armaTracerCor == "Arco-iris" then
            colorValue.Color = Color3.fromHSV((os.clock() * 0.25) % 1, 1, 1)
            return
        end

        for _, value in ipairs(TRACER_COLORS) do
            if value[1] == State.armaTracerCor and value[2] then
                colorValue.Color = value[2]
                return
            end
        end
    end

    Runtime.restaurarArmas = function()
        pcall(applyTracerColor)
        local colorValue = getBulletColorValue()

        if colorValue and originalBulletColor then
            pcall(function()
                colorValue.Color = originalBulletColor
            end)
        end
    end

    local tracerTimer = 0

    trackConnection(RunService.Heartbeat:Connect(function(deltaTime)
        if Runtime.destroyed or not State.armaTracer then
            return
        end

        tracerTimer = tracerTimer + deltaTime

        if tracerTimer < 0.15 then
            return
        end

        tracerTimer = 0
        pcall(applyTracerColor)
    end))
end

local function onHeartbeat(deltaTime)
    if Runtime.destroyed then
        return
    end

    local humanoid = getHumanoid()

    if Runtime.pumpkinBelowActive or Runtime.fixedTeleportBusy then
        return
    end

    pcall(Runtime.updateJumpPower, humanoid)

    if Runtime.travelBusy then
        return
    end

    if Runtime.robbery and (Runtime.robbery.enabled or Runtime.robbery.paused) then
        return
    end
end

trackConnection(RunService.Heartbeat:Connect(onHeartbeat))

trackConnection(RunService.Stepped:Connect(function()
    if not Runtime.destroyed then
        pcall(Runtime.noclip.step)
    end
end))

do
    local claiming = false
    local lastStatus = {}

    local function syncToggle(control, value)
        if not control then
            return
        end

        if control:Get() ~= value then
            control:Set(value, true)
        end
    end

    local function syncStatus(key, value)
        local control = Runtime[key]

        if control and lastStatus[key] ~= value then
            lastStatus[key] = value
            control:Set(value, true)
        end
    end

    function Runtime.syncFarmControls()
        syncToggle(Runtime.coletaControl, State.autoColetaEnabled)
        syncToggle(Runtime.lixeiroControl, State.autoFarmLixeiroEnabled)

    function Runtime.claimFarm(feature)
        if Runtime.destroyed then
            return false
        end

        if Runtime.fixedTeleportBusy and feature ~= "teleport" then
            notify("Aguarde o teleporte terminar.")
            return false
        end

        if claiming then
            return true
        end

        claiming = true

        if feature ~= "items" and State.autoColetaEnabled and Runtime.stopAutoColeta then
            pcall(Runtime.stopAutoColeta)
        end

        if feature ~= "garbage" and State.autoFarmLixeiroEnabled and Runtime.startLixeiro then
            pcall(Runtime.startLixeiro, false)
        end

        if feature ~= "fishing" and State.autoFarmPescaEnabled and Runtime.startPesca then
            pcall(Runtime.startPesca, false)
        end

        if Runtime.travel then
            Runtime.travel.cancel()
        end

        if Runtime.stopMovement then
            Runtime.stopMovement()
        end

        claiming = false
        Runtime.syncFarmControls()
        return true
    end

    local function wrapFarmStarter(name, feature, always)
        local original = Runtime[name]

        if not original then

        Runtime[name] = function(enabled, ...)
            if (always or enabled == true) and not claiming then
                if not Runtime.claimFarm(feature) then
                    return
                end
            end

            return original(enabled, ...)
        end

    local function readCurrentValue(entry)
        local robberyKey = entry.key:match("^robbery%.(.+)$")

    local function sanitize(entry, value)
        if entry.kind == "Toggle" then
            if type(value) == "boolean" then
                return value
            end
        elseif entry.kind == "Slider" then
            if type(value) == "number" and value == value and math.abs(value) < math.huge then
                local clamped = math.clamp(value, entry.min, entry.max)
                local step = entry.step
                return math.clamp(math.round((clamped - entry.min) / step) * step + entry.min, entry.min, entry.max)
            end
        elseif entry.kind == "Dropdown" and type(value) == "string" then
            if table.find(entry.options, value) then
                return value
            end
        end

        return nil
    end

    function Runtime.configControl(section, kind, key, options)
        assert(not Config.byKey[key], "Preferência duplicada: " .. key)
        local callback = options.Callback

        options.Callback = function(value, ...)
            if Runtime.destroyed then
                return
            end

            if not Config.applying then
                Config.cancelPendingLoad()
            end

            return callback(value, ...)
        end

        local control = section[kind](section, options)

        local entry = {
            key = key,
            kind = kind,
            control = control,
            callback = callback,
            default = options.Default,
            min = options.Min,
            max = options.Max,
            step = options.Step or 1,
            options = options.Options,
        }

        if kind == "Toggle" then
            entry.default = false

        Config.entries[#Config.entries + 1], Config.byKey[key] = entry, entry

    end

    local function normalizeSettings(settings)

        end

        return result
    end

    function Config.sync()
        for _, value in ipairs(Config.entries) do
            local value2 = sanitize(value, readCurrentValue(value))

            if value2 ~= nil and value.control:Get() ~= value2 then
                value.control:Set(value2, true)
            end
        end
    end

    function Config.collect()
        local values = {}

        for _, value in ipairs(Config.entries) do
            local value2 = sanitize(value, readCurrentValue(value))

            if value2 == nil then
                value2 = value.default
            end

            values[value.key] = value2
        end

        return normalizeSettings(values)
    end

    function Config.apply(settings)
        local values, failures = normalizeSettings(settings), 0
        Config.applying = true

        local function applyEntry(entry)
            local value = values[entry.key]
            entry.control:Set(value, true)
            local success = pcall(entry.callback, value)

            if not success then
                failures = failures + 1
            end
        end

        for _, value in ipairs(Config.entries) do
            if value.kind ~= "Toggle" then
                applyEntry(value)
            end
        end

        for _, value in ipairs(Config.entries) do
            if value.kind == "Toggle" and not DEFERRED_KEYS[value.key] then
                applyEntry(value)
            end
        end

        for _, value in ipairs(Config.entries) do
            if DEFERRED_KEYS[value.key] and not table.find(EXCLUSIVE_FARMS, value.key) then
                applyEntry(value)
            end
        end

        for _, value in ipairs(EXCLUSIVE_FARMS) do
            if Config.byKey[value] and not values[value] then
                applyEntry(Config.byKey[value])
            end
        end

        for _, value in ipairs(EXCLUSIVE_FARMS) do
            if Config.byKey[value] and values[value] then
                applyEntry(Config.byKey[value])
            end
        end

        Config.applying = false
        Config.sync()
        return failures == 0, failures
    end

    local function parseConfig(raw)
        if type(raw) ~= "string" or #raw > 65536 then
            return nil
        end

        local success, result = pcall(HttpService.JSONDecode, HttpService, raw)

        if not success or type(result) ~= "table" or result.version ~= 1 or result.userId ~= USER_ID or result.gameId ~= GAME_ID or type(result.settings) ~= "table" then
            return nil
        end

        if result.movementVersion ~= 12 then
            result.settings.flySpeed, result.settings.spoofWalkSpeed = 1.5, 5
        end

        return normalizeSettings(result.settings)
    end

    local function readConfigFile(path)
        if type(fs.read) ~= "function" then
            return nil, "unsupported"
        end

        if type(fs.exists) == "function" then
            local success, result = pcall(fs.exists, path)

            if success and not result then
                return nil, "missing"
            end
        end

        local success, result = pcall(fs.read, path)

        if not success then
            return nil, "read"
        end

        return result, nil
    end

    function Config.save()
        if Runtime.destroyed then
            return false
        end

        Config.cancelPendingLoad()

        if type(fs.read) ~= "function" or type(fs.write) ~= "function" then
            setConfigStatus("Este executor não oferece leitura e gravação de configurações.")
            notify(Config.status)
            return false
        end

        local success, result = pcall(HttpService.JSONEncode, HttpService, {
            version = 1,
            movementVersion = 12,
            userId = USER_ID,
            gameId = GAME_ID,
            settings = Config.collect(),
        })

        if not success then
            setConfigStatus("Não foi possível preparar a configuração.")
            notify(Config.status)
            return false
        end

        if type(fs.makeFolder) == "function" then
            local folderExists = false

            if type(fs.isFolder) == "function" then
                local success2, result2 = pcall(fs.isFolder, CONFIG_FOLDER)
                folderExists = success2 and result2 == true
            end

            if not folderExists then
                pcall(fs.makeFolder, CONFIG_FOLDER)
            end
        end

        local previous = readConfigFile(CONFIG_PATH)

        if previous and parseConfig(previous) then
            if not pcall(fs.write, CONFIG_PATH .. ".bak", previous) then
                setConfigStatus("Falha ao guardar a cópia anterior. Configuração preservada.")
                notify(Config.status)
                return false
            end
        else
            previous = nil
        end

        local success2 = pcall(fs.write, CONFIG_PATH, result)
        local verify = success2 and readConfigFile(CONFIG_PATH)

        if not success2 or verify ~= result then
            if previous then
                pcall(fs.write, CONFIG_PATH, previous)
            end

            setConfigStatus("Não foi possível confirmar a gravação. Tente salvar novamente.")
            notify(Config.status)
            return false
        end

        if not previous then
            pcall(fs.write, CONFIG_PATH .. ".bak", result)
        end

        setConfigStatus("Configuração salva. Será carregada automaticamente nesta conta e neste jogo.")
        notify("Configuração salva com sucesso.")
        return true
    end

    function Config.boot()
        if Config.booted or Runtime.destroyed then
            return
        end

        Config.booted = true
        local revision = Config.revision
        local raw, readError = readConfigFile(CONFIG_PATH)
        local settings = raw and parseConfig(raw)
        local fromBackup = false

        if not settings and readError ~= "unsupported" then
            local backup = readConfigFile(CONFIG_PATH .. ".bak")
            settings = backup and parseConfig(backup)
            fromBackup = settings ~= nil
        end

        if not settings then
            if readError == "unsupported" then
                setConfigStatus("Tudo desligado. Leitura de configuração indisponível neste executor.")
            elseif raw then
                setConfigStatus("Configuração inválida. Tudo inicia desligado; salve uma nova configuração.")
            elseif readError == "read" then
                setConfigStatus("Não foi possível ler a configuração. Tudo inicia desligado.")
            else
                setConfigStatus("Nenhuma configuração salva. Tudo inicia desligado.")
            end

            return
        end

        local needsCharacter = false

        for key in pairs(NEEDS_CHARACTER) do
            if settings[key] then
                needsCharacter = true
                break
            end
        end

        if Runtime.destroyed or Config.revision ~= revision then
            return
        end

        Config.pending = true

        if needsCharacter then
            setConfigStatus("Configuração encontrada. Aguardando o personagem carregar...")

            while not Runtime.destroyed and Config.revision == revision do
                local humanoid, rootPart = getHumanoid(), getRootPart()

                if rootPart and humanoid and humanoid.Health > 0 then
                    break
                end

                task.wait(0.25)
            end
        end

        if Runtime.destroyed or Config.revision ~= revision then
            return
        end

        Config.pending = false
        local ok = Config.apply(settings)

        if ok then
            setConfigStatus(fromBackup and "Configuração recuperada da cópia anterior. Salve para confirmar." or "Configuração salva carregada.")
        else
            setConfigStatus("Preferências carregadas; uma função não iniciou. Confira os toggles antes de salvar.")
        end
    end
end
